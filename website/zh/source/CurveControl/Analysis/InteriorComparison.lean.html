3/-!
4# Exponential comparison with derivatives only at interior times
5
6The integrating-factor theorem already proved in `EnergyIntegral` supplies a
7comparison that needs no initial-time derivative. This is the interface used
8when geometric evolution is proved on the interior and continuity is available
9up to the endpoints. The differential inequalities are explicit analysis inputs.

16theorem le_exp_of_deriv_interior_le
17 {f f' : ℝ → ℝ} {a b K : ℝ} (hab : a ≤ b)
18 (hf : ContinuousOn f (Icc a b))
19 (hf' : ∀ t ∈ Ioo a b, HasDerivWithinAt f (f' t) (Ioi t) t)
20 (hbound : ∀ t ∈ Ioo a b, f' t ≤ K * f t) :
21 f b ≤ f a * exp (K * (b - a)) := by
22 have he : IntegrableOn (fun _ : ℝ => (0 : ℝ)) (Icc a b) := by simp
23 have h := weighted_energy_integral_le (E := fun _ => 0) hab hf hf' he
24 (fun t ht => by simpa using hbound t ht)
25 have hw : exp (-K * (b - a)) * f b ≤ f a := by
26 simpa only [mul_zero, intervalIntegral.integral_zero, sub_nonneg] using h
27 have hexp : exp (K * (b - a)) * exp (-K * (b - a)) = 1 := by
28 rw [← exp_add]
29 have hz : K * (b - a) + -K * (b - a) = 0 := by ring
30 rw [hz, exp_zero]
31 calc
32 f b = exp (K * (b - a)) * (exp (-K * (b - a)) * f b) := by
33 rw [← mul_assoc, hexp, one_mul]
34 _ ≤ exp (K * (b - a)) * f a :=
35 mul_le_mul_of_nonneg_left hw (exp_pos _).le
36 _ = f a * exp (K * (b - a)) := mul_comm _ _

38theorem coupled_exp_comparison_interior
39 {L Q L' Q' : ℝ → ℝ} {a b c₁ c₂ : ℝ} (hab : a ≤ b)
40 (hL : ContinuousOn L (Icc a b)) (hQ : ContinuousOn Q (Icc a b))
41 (hL' : ∀ t ∈ Ioo a b, HasDerivWithinAt L (L' t) (Ioi t) t)
42 (hQ' : ∀ t ∈ Ioo a b, HasDerivWithinAt Q (Q' t) (Ioi t) t)
43 (hLbound : ∀ t ∈ Ioo a b, L' t ≤ c₂ * L t)
44 (hQbound : ∀ t ∈ Ioo a b, Q' t ≤ (c₁ + c₂) * Q t + c₁ * L t) :
45 L b ≤ L a * exp (c₂ * (b - a)) ∧
46 Q b + L b ≤ (Q a + L a) * exp ((c₁ + c₂) * (b - a)) := by
47 constructor
48 · exact le_exp_of_deriv_interior_le hab hL hL' hLbound
49 · apply le_exp_of_deriv_interior_le hab (hQ.add hL)
50 (fun t ht => (hQ' t ht).add (hL' t ht))
51 intro t ht
52 calc
53 Q' t + L' t ≤ ((c₁ + c₂) * Q t + c₁ * L t) + c₂ * L t :=
54 add_le_add (hQbound t ht) (hLbound t ht)
55 _ = (c₁ + c₂) * (Q t + L t) := by ring

57theorem coupled_exp_comparison_interior_on
58 {L Q L' Q' : ℝ → ℝ} {a b c₁ c₂ : ℝ}
59 (hL : ContinuousOn L (Icc a b)) (hQ : ContinuousOn Q (Icc a b))
60 (hL' : ∀ t ∈ Ioo a b, HasDerivWithinAt L (L' t) (Ioi t) t)
61 (hQ' : ∀ t ∈ Ioo a b, HasDerivWithinAt Q (Q' t) (Ioi t) t)
62 (hLbound : ∀ t ∈ Ioo a b, L' t ≤ c₂ * L t)
63 (hQbound : ∀ t ∈ Ioo a b, Q' t ≤ (c₁ + c₂) * Q t + c₁ * L t) :
64 ∀ t ∈ Icc a b,
65 L t ≤ L a * exp (c₂ * (t - a)) ∧
66 Q t + L t ≤ (Q a + L a) * exp ((c₁ + c₂) * (t - a)) := by
67 intro t ht
68 have hi : Icc a t ⊆ Icc a b := Icc_subset_Icc le_rfl ht.2
69 have ho : Ioo a t ⊆ Ioo a b := Ioo_subset_Ioo le_rfl ht.2
70 exact coupled_exp_comparison_interior ht.1 (hL.mono hi) (hQ.mono hi)
71 (fun r hr => hL' r (ho hr)) (fun r hr => hQ' r (ho hr))
72 (fun r hr => hLbound r (ho hr)) (fun r hr => hQbound r (ho hr))
