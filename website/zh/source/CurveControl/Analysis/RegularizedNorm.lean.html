5/-!
6# Scalar regularization at zero curvature
7
8The regularization of a squared norm `q` is `sqrt (q + ε²)`.
9These scalar lemmas do not assume or assert a geometric evolution equation.
10-/

14noncomputable def regularizedNorm (q ε : ℝ) : ℝ := Real.sqrt (q + ε ^ 2)
15
16theorem regularizedNorm_nonneg (q ε : ℝ) : 0 ≤ regularizedNorm q ε :=
17 Real.sqrt_nonneg _
18
19theorem regularizedNorm_pos {q ε : ℝ} (hq : 0 ≤ q) (hε : 0 < ε) :
20 0 < regularizedNorm q ε := by
21 apply Real.sqrt_pos.2
22 exact add_pos_of_nonneg_of_pos hq (sq_pos_of_pos hε)

24theorem regularizedNorm_sq {q ε : ℝ} (hq : 0 ≤ q) :
25 regularizedNorm q ε ^ 2 = q + ε ^ 2 :=
26 Real.sq_sqrt (add_nonneg hq (sq_nonneg ε))

28theorem sqrt_le_regularizedNorm (q ε : ℝ) :
29 Real.sqrt q ≤ regularizedNorm q ε :=
30 Real.sqrt_le_sqrt (le_add_of_nonneg_right (sq_nonneg ε))

32theorem regularizedNorm_le_sqrt_add {q ε : ℝ} (hq : 0 ≤ q) (hε : 0 ≤ ε) :
33 regularizedNorm q ε ≤ Real.sqrt q + ε := by
34 have hs := Real.sq_sqrt hq
35 have hr := regularizedNorm_sq (ε := ε) hq
36 have hn := regularizedNorm_nonneg q ε
37 have ht := Real.sqrt_nonneg q
38 nlinarith [mul_nonneg ht hε]

40theorem regularizedNorm_sub_sqrt_bounds {q ε : ℝ} (hq : 0 ≤ q) (hε : 0 ≤ ε) :
41 0 ≤ regularizedNorm q ε - Real.sqrt q ∧
42 regularizedNorm q ε - Real.sqrt q ≤ ε := by
43 constructor
44 · exact sub_nonneg.mpr (sqrt_le_regularizedNorm q ε)
45 · exact sub_le_iff_le_add.mpr (by simpa [add_comm] using regularizedNorm_le_sqrt_add hq hε)

47theorem sq_div_regularizedNorm_le_sqrt_cube {q ε : ℝ} (hq : 0 ≤ q) (hε : 0 < ε) :
48 q ^ 2 / regularizedNorm q ε ≤ (Real.sqrt q) ^ 3 := by
49 apply (div_le_iff₀ (regularizedNorm_pos hq hε)).2
50 have hs := Real.sq_sqrt hq
51 have hm := mul_le_mul_of_nonneg_left (sqrt_le_regularizedNorm q ε)
52 (pow_nonneg (Real.sqrt_nonneg q) 3)
53 calc
54 q ^ 2 = ((Real.sqrt q) ^ 2) ^ 2 := by rw [hs]
55 _ = (Real.sqrt q) ^ 3 * Real.sqrt q := by ring
56 _ ≤ (Real.sqrt q) ^ 3 * regularizedNorm q ε := hm

58theorem div_regularizedNorm_le {q ε : ℝ} (hq : 0 ≤ q) (hε : 0 < ε) :
59 q / regularizedNorm q ε ≤ regularizedNorm q ε := by
60 apply (div_le_iff₀ (regularizedNorm_pos hq hε)).2
61 nlinarith [regularizedNorm_sq (ε := ε) hq, sq_nonneg ε]

63theorem sqrt_div_regularizedNorm_le_one {q ε : ℝ} (hq : 0 ≤ q) (hε : 0 < ε) :
64 Real.sqrt q / regularizedNorm q ε ≤ 1 := by
65 apply (div_le_iff₀ (regularizedNorm_pos hq hε)).2
66 simpa using sqrt_le_regularizedNorm q ε

68theorem hasDerivAt_regularizedNorm {q : ℝ → ℝ} {q' t ε : ℝ}
69 (hd : HasDerivAt q q' t) (hq : 0 ≤ q t) (hε : 0 < ε) :
70 HasDerivAt (fun s => regularizedNorm (q s) ε)
71 (q' / (2 * regularizedNorm (q t) ε)) t := by
72 have ha : HasDerivAt (fun s => q s + ε ^ 2) q' t := hd.add_const (ε ^ 2)
