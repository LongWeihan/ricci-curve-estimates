4/-!
5# Intrinsic ambient tensor bounds and their curve evaluations
6
7The norms in these bounds are explicitly computed from the evolving metric.
8They are not the fixed model-space norm. The constants bound the actual
9canonical Ricci tensor, curvature form, and covariant derivative of Ricci.
10Compactness-based existence of uniform constants is a separate obligation.
11-/

25/-- **Math.** The norm induced by the actual metric on this tangent fiber. -/
26def metricNorm (g : Riemannian.RiemannianMetric I M) (p : M) (v : TangentSpace I p) : ℝ :=
27 Real.sqrt (g.metricInner p v v)
28
29theorem metricNorm_nonneg (g : Riemannian.RiemannianMetric I M) (p : M) (v : TangentSpace I p) :
30 0 ≤ metricNorm g p v := Real.sqrt_nonneg _
31
32theorem metricNorm_sq (g : Riemannian.RiemannianMetric I M) (p : M) (v : TangentSpace I p) :
33 metricNorm g p v ^ 2 = g.metricInner p v v :=
34 Real.sq_sqrt (g.metricInner_self_nonneg p v)
35
36theorem metricNorm_unit (g : Riemannian.RiemannianMetric I M) {p : M} {v : TangentSpace I p}
37 (hv : g.metricInner p v v = 1) : metricNorm g p v = 1 := by
38 simp only [metricNorm, hv, Real.sqrt_one]

40/-- **Math.** The canonical Levi-Civita covariant derivative of Ricci, without
41an independent tensor or connection supplied by the caller. -/
42def covRic (g : Riemannian.RiemannianMetric I M) (p : M)
43 (u v w : TangentSpace I p) : ℝ :=
44 covRicciAt g g.leviCivitaConnection
45 (g.leviCivitaConnection.isLeviCivita_of_koszulDual g
46 (fun X Y W q => g.koszulDualSection_dual X Y W q)) p u v w

48/-- **Math.** The actual canonical curvature four-form, in upstream slot order. -/
49def rm (g : Riemannian.RiemannianMetric I M) (p : M)
50 (u v w z : TangentSpace I p) : ℝ :=
51 curvatureFormAt g g.leviCivitaConnection p u v w z

53/-- **Math.** Intrinsic multilinear tensor evaluation bounds at one point.
54Every slot is measured with `g`, not with the model inner product. These are
55ambient hypotheses only; none of the curve evolution conclusions is a field. -/
56structure TensorBoundsAt (g : Riemannian.RiemannianMetric I M) (p : M)
57 (B K D : ℝ) : Prop where
58 ricci_nonneg : 0 ≤ B
59 curvature_nonneg : 0 ≤ K
60 covRic_nonneg : 0 ≤ D
61 ricci : ∀ u v : TangentSpace I p,
62 |ricciTensorAt g p u v| ≤ B * metricNorm g p u * metricNorm g p v
63 curvature : ∀ u v w z : TangentSpace I p,
64 |rm g p u v w z| ≤ K * metricNorm g p u * metricNorm g p v *
65 metricNorm g p w * metricNorm g p z
66 covRic : ∀ u v w : TangentSpace I p,
67 |AmbientBounds.covRic g p u v w| ≤ D * metricNorm g p u * metricNorm g p v *
68 metricNorm g p w

70/-- **Math.** A single set of constants controls all spatial points and times. -/
71def UniformTensorBoundsOn (g : ℝ → Riemannian.RiemannianMetric I M) (J : Set ℝ)
72 (B K D : ℝ) : Prop := ∀ t ∈ J, ∀ p, TensorBoundsAt (g t) p B K D

79theorem ricci_unit {S : TangentSpace I p} (hS : g.metricInner p S S = 1) :
80 |ricciTensorAt g p S S| ≤ B := by
81 simpa only [metricNorm_unit g hS, mul_one] using h.ricci S S
82
83theorem ricci_diagonal (V : TangentSpace I p) :
84 |ricciTensorAt g p V V| ≤ B * metricNorm g p V ^ 2 := by
85 simpa only [pow_two, mul_assoc] using h.ricci V V
86
87theorem rm_unit {S : TangentSpace I p} (hS : g.metricInner p S S = 1)
88 (V : TangentSpace I p) :
89 |rm g p V S V S| ≤ K * metricNorm g p V ^ 2 := by
90 simpa only [metricNorm_unit g hS, mul_one, pow_two, mul_assoc] using h.curvature V S V S

92theorem covRic_unit_unit {S : TangentSpace I p} (hS : g.metricInner p S S = 1)
93 (V : TangentSpace I p) :
94 |AmbientBounds.covRic g p S S V| ≤ D * metricNorm g p V := by
95 simpa only [metricNorm_unit g hS, mul_one] using h.covRic S S V

97theorem covRic_unit_last {S : TangentSpace I p} (hS : g.metricInner p S S = 1)
98 (V : TangentSpace I p) :
99 |AmbientBounds.covRic g p V S S| ≤ D * metricNorm g p V := by
100 simpa only [metricNorm_unit g hS, mul_one] using h.covRic V S S

102/-- **Math.** The quadratic ambient contribution in the squared-curvature
103identity is bounded by `(6B+2K) k²`. -/
104theorem quadratic_error {S : TangentSpace I p} (hS : g.metricInner p S S = 1)
105 (V : TangentSpace I p) :
106 |4 * metricNorm g p V ^ 2 * ricciTensorAt g p S S -
107 2 * ricciTensorAt g p V V + 2 * rm g p V S V S| ≤
108 (6 * B + 2 * K) * metricNorm g p V ^ 2 := by
109 have h1 : |4 * metricNorm g p V ^ 2 * ricciTensorAt g p S S| ≤
110 4 * metricNorm g p V ^ 2 * B := by
111 rw [abs_mul, abs_of_nonneg (mul_nonneg (by norm_num) (sq_nonneg _))]
112 exact mul_le_mul_of_nonneg_left (h.ricci_unit hS) (mul_nonneg (by norm_num) (sq_nonneg _))
113 have h2 : |2 * ricciTensorAt g p V V| ≤ 2 * (B * metricNorm g p V ^ 2) := by
114 rw [abs_mul, abs_of_pos (by norm_num : (0 : ℝ) < 2)]
115 exact mul_le_mul_of_nonneg_left (h.ricci_diagonal V) (by norm_num)
116 have h3 : |2 * rm g p V S V S| ≤ 2 * (K * metricNorm g p V ^ 2) := by
117 rw [abs_mul, abs_of_pos (by norm_num : (0 : ℝ) < 2)]
118 exact mul_le_mul_of_nonneg_left (h.rm_unit hS V) (by norm_num)
119 calc
120 _ ≤ |4 * metricNorm g p V ^ 2 * ricciTensorAt g p S S - 2 * ricciTensorAt g p V V| +
121 |2 * rm g p V S V S| := abs_add_le _ _
122 _ ≤ (|4 * metricNorm g p V ^ 2 * ricciTensorAt g p S S| +
123 |2 * ricciTensorAt g p V V|) + |2 * rm g p V S V S| :=
124 add_le_add (by
125 simpa only [sub_zero, zero_sub, abs_neg] using
126 (abs_sub_le (4 * metricNorm g p V ^ 2 * ricciTensorAt g p S S) (0 : ℝ)
127 (2 * ricciTensorAt g p V V))) le_rfl
128 _ ≤ (4 * metricNorm g p V ^ 2 * B + 2 * (B * metricNorm g p V ^ 2)) +
129 2 * (K * metricNorm g p V ^ 2) := add_le_add (add_le_add h1 h2) h3
130 _ = _ := by ring

132/-- **Math.** The covariant-Ricci contribution is linear in curvature norm. -/
133theorem linear_error {S : TangentSpace I p} (hS : g.metricInner p S S = 1)
134 (V : TangentSpace I p) :
135 |-4 * AmbientBounds.covRic g p S S V + 2 * AmbientBounds.covRic g p V S S| ≤
136 6 * D * metricNorm g p V := by
137 calc
138 _ ≤ |-4 * AmbientBounds.covRic g p S S V| + |2 * AmbientBounds.covRic g p V S S| := abs_add_le _ _
139 _ = 4 * |AmbientBounds.covRic g p S S V| + 2 * |AmbientBounds.covRic g p V S S| := by
140 simp only [abs_mul]; norm_num
141 _ ≤ 4 * (D * metricNorm g p V) + 2 * (D * metricNorm g p V) :=
142 add_le_add (mul_le_mul_of_nonneg_left (h.covRic_unit_unit hS V) (by norm_num))
143 (mul_le_mul_of_nonneg_left (h.covRic_unit_last hS V) (by norm_num))
144 _ = _ := by ring
145end TensorBoundsAt

152/-- **Math.** Intrinsic environmental tensor bounds evaluated on the actual
153unit tangent and actual curvature vector of an immersed curve. -/
154theorem curve_evaluations {g : Riemannian.RiemannianMetric I M} (γ : ℝ → M) (x : ℝ)
155 {B K D : ℝ} (h : TensorBoundsAt g (γ x) B K D)
156 (himm : curveVelocity (I := I) γ x ≠ 0) :
157 let S := unitTangent g γ x
158 let V := curvatureVector g γ x
159 |ricciTensorAt g (γ x) S S| ≤ B ∧
160 |ricciTensorAt g (γ x) V V| ≤ B * curvature g γ x ^ 2 ∧
161 |rm g (γ x) V S V S| ≤ K * curvature g γ x ^ 2 ∧
162 |covRic g (γ x) S S V| ≤ D * curvature g γ x ∧
163 |covRic g (γ x) V S S| ≤ D * curvature g γ x := by
164 have hS := unitTangent_unit g γ x himm
165 exact ⟨h.ricci_unit hS, h.ricci_diagonal _, h.rm_unit hS _,
166 h.covRic_unit_unit hS _, h.covRic_unit_last hS _⟩

168/-- **Math.** The actual curve's ambient q-evolution error, with constants
169independent of the selected curve whenever the ambient tensor bounds are. -/
170theorem curve_ambient_error {g : Riemannian.RiemannianMetric I M} (γ : ℝ → M) (x : ℝ)
171 {B K D : ℝ} (h : TensorBoundsAt g (γ x) B K D)
172 (himm : curveVelocity (I := I) γ x ≠ 0) :
173 let S := unitTangent g γ x
174 let V := curvatureVector g γ x
175 |(4 * curvatureSq g γ x * ricciTensorAt g (γ x) S S -
176 2 * ricciTensorAt g (γ x) V V + 2 * rm g (γ x) V S V S) +
177 (-4 * covRic g (γ x) S S V + 2 * covRic g (γ x) V S S)| ≤
178 (6 * B + 2 * K) * curvatureSq g γ x + 6 * D * curvature g γ x := by
179 dsimp only
180 have hS := unitTangent_unit g γ x himm
181 have hq := h.quadratic_error hS (curvatureVector g γ x)
182 have hl := h.linear_error hS (curvatureVector g γ x)
183 simp only [metricNorm_curvatureVector, curvature_sq] at hq hl
184 exact (abs_add_le _ _).trans (add_le_add hq hl)
