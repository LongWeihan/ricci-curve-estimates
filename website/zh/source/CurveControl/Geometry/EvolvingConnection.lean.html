6# Time-dependent connection coefficients in a fixed chart
7
8Adapted from MorganTianLib.Ch01.CurvatureCommutation: the coefficient field may
9now depend on the parameter separately from the chart point. Product rules and
10Schwarz symmetry prove all commutation identities below. No curvature evolution
11or commutation conclusion is assumed. This is the coordinate analytic layer;
12identification with the Levi-Civita connection of a Ricci-flow metric is separate.

20/-- Connection along a map, with an independently varying coefficient field. -/
21def covDerivAlong (Γ : P → E →L[ℝ] E →L[ℝ] E) (u V : P → E) (d p : P) : E :=
22 fderiv ℝ V p d + Γ p (fderiv ℝ u p d) (V p)

24theorem covDerivAlong_def (Γ : P → E →L[ℝ] E →L[ℝ] E) (u V : P → E) (d p : P) :
25 covDerivAlong Γ u V d p = fderiv ℝ V p d + Γ p (fderiv ℝ u p d) (V p) := rfl

27/-- Actual derivative expansion, including the derivative of the coefficients. -/
28theorem fderiv_covDerivAlong_apply {Γ : P → E →L[ℝ] E →L[ℝ] E} {u V : P → E}
29 {p : P} (hu : ContDiffAt ℝ 2 u p) (hV : ContDiffAt ℝ 2 V p)
30 (hΓ : DifferentiableAt ℝ Γ p) (d e : P) :
31 fderiv ℝ (covDerivAlong Γ u V d) p e
32 = fderiv ℝ (fderiv ℝ V) p e d
33 + fderiv ℝ Γ p e (fderiv ℝ u p d) (V p)
34 + Γ p (fderiv ℝ (fderiv ℝ u) p e d) (V p)
35 + Γ p (fderiv ℝ u p d) (fderiv ℝ V p e) := by
36 have h21 : ((1 : ℕ∞ω) + 1 : ℕ∞ω) ≤ 2 := by norm_num
37 have hu1 : DifferentiableAt ℝ u p := hu.differentiableAt (by norm_num)
38 have hV1 : DifferentiableAt ℝ V p := hV.differentiableAt (by norm_num)
39 have hD2u : HasFDerivAt (fderiv ℝ u) (fderiv ℝ (fderiv ℝ u) p) p :=
40 ((hu.fderiv_right h21).differentiableAt (by norm_num)).hasFDerivAt
41 have hD2V : HasFDerivAt (fderiv ℝ V) (fderiv ℝ (fderiv ℝ V) p) p :=
42 ((hV.fderiv_right h21).differentiableAt (by norm_num)).hasFDerivAt
43 -- the two directional-derivative fields
44 have happV : HasFDerivAt (fun q => fderiv ℝ V q d)
45 ((fderiv ℝ (fderiv ℝ V) p).flip d) p := by
46 have h := hD2V.clm_apply (hasFDerivAt_const d p)
47 simpa using h
48 have happu : HasFDerivAt (fun q => fderiv ℝ u q d)
49 ((fderiv ℝ (fderiv ℝ u) p).flip d) p := by
50 have h := hD2u.clm_apply (hasFDerivAt_const d p)
51 simpa using h
52 -- the Christoffel term, by two applications of the CLM product rule
53 have hA : HasFDerivAt (fun q => Γ q (fderiv ℝ u q d))
54 ((Γ p).comp ((fderiv ℝ (fderiv ℝ u) p).flip d)
55 + (fderiv ℝ Γ p).flip (fderiv ℝ u p d)) p :=
56 hΓ.hasFDerivAt.clm_apply happu
57 have hG := hA.clm_apply hV1.hasFDerivAt
58 have htot : HasFDerivAt (covDerivAlong Γ u V d)
59 (((fderiv ℝ (fderiv ℝ V) p).flip d)
60 + ((Γ p (fderiv ℝ u p d)).comp (fderiv ℝ V p)
61 + ((Γ p).comp ((fderiv ℝ (fderiv ℝ u) p).flip d)
62 + (fderiv ℝ Γ p).flip
63 (fderiv ℝ u p d)).flip (V p))) p := by
64 exact happV.add hG
65 rw [htot.fderiv]
66 simp only [add_apply, ContinuousLinearMap.coe_comp,
67 Function.comp_apply, ContinuousLinearMap.flip_apply]
68 abel

70/-- Commutator for a parameter-dependent coefficient field. -/
71theorem covDerivAlong_comm {Γ : P → E →L[ℝ] E →L[ℝ] E} {u V : P → E} {p : P}
72 (hu : ContDiffAt ℝ 2 u p) (hV : ContDiffAt ℝ 2 V p)
73 (hΓ : DifferentiableAt ℝ Γ p) (d₁ d₂ : P) :
74 covDerivAlong Γ u (covDerivAlong Γ u V d₂) d₁ p
75 - covDerivAlong Γ u (covDerivAlong Γ u V d₁) d₂ p
76 = fderiv ℝ Γ p d₁ (fderiv ℝ u p d₂) (V p)
77 - fderiv ℝ Γ p d₂ (fderiv ℝ u p d₁) (V p)
78 + Γ p (fderiv ℝ u p d₁) (Γ p (fderiv ℝ u p d₂) (V p))
79 - Γ p (fderiv ℝ u p d₂) (Γ p (fderiv ℝ u p d₁) (V p)) := by
80 have h1 := fderiv_covDerivAlong_apply hu hV hΓ d₂ d₁
81 have h2 := fderiv_covDerivAlong_apply hu hV hΓ d₁ d₂
82 have hVs := (hV.isSymmSndFDerivAt (by simp)).eq d₂ d₁
83 have hus := (hu.isSymmSndFDerivAt (by simp)).eq d₂ d₁
84 rw [covDerivAlong_def Γ u (covDerivAlong Γ u V d₂) d₁ p,
85 covDerivAlong_def Γ u (covDerivAlong Γ u V d₁) d₂ p, h1, h2,
86 covDerivAlong_def Γ u V d₂ p, covDerivAlong_def Γ u V d₁ p]
87 simp only [map_add]
88 rw [hVs, hus]
89 abel

91/-- Torsion-free mixed velocities, without assuming mixed derivatives commute. -/
92theorem covDerivAlong_fderiv_symm {Γ : P → E →L[ℝ] E →L[ℝ] E} {u : P → E}
93 {p : P} (hu : ContDiffAt ℝ 2 u p)
94 (hΓsymm : ∀ X Y, Γ p X Y = Γ p Y X) (d₁ d₂ : P) :
95 covDerivAlong Γ u (fun q => fderiv ℝ u q d₂) d₁ p
96 = covDerivAlong Γ u (fun q => fderiv ℝ u q d₁) d₂ p := by
97 have h21 : ((1 : ℕ∞ω) + 1 : ℕ∞ω) ≤ 2 := by norm_num
98 have hD2u : HasFDerivAt (fderiv ℝ u) (fderiv ℝ (fderiv ℝ u) p) p :=
99 ((hu.fderiv_right h21).differentiableAt (by norm_num)).hasFDerivAt
100 have happ : ∀ d : P, fderiv ℝ (fun q => fderiv ℝ u q d) p
101 = (fderiv ℝ (fderiv ℝ u) p).flip d := by
102 intro d
103 have h := hD2u.clm_apply (hasFDerivAt_const d p)
104 have h' : HasFDerivAt (fun q => fderiv ℝ u q d)
105 ((fderiv ℝ (fderiv ℝ u) p).flip d) p := by simpa using h
106 exact h'.fderiv
107 rw [covDerivAlong_def, covDerivAlong_def, happ d₁, happ d₂]
108 simp only [ContinuousLinearMap.flip_apply]
109 rw [(hu.isSymmSndFDerivAt (by simp)).eq d₁ d₂,
110 hΓsymm (fderiv ℝ u p d₁) (fderiv ℝ u p d₂)]

113/-- The time-first pullback of an evolving Christoffel contraction. -/
114def coefficients (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (u : ℝ × ℝ → E)
115 (p : ℝ × ℝ) : E →L[ℝ] E →L[ℝ] E := Γ p.1 (u p)
116
117/-- Time and spatial covariant derivatives use actual Fréchet derivatives. -/
118def covTime (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (u V : ℝ × ℝ → E) :=
119 covDerivAlong (coefficients Γ u) u V (1, 0)
120
121def covSpace (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (u V : ℝ × ℝ → E) :=
122 covDerivAlong (coefficients Γ u) u V (0, 1)

124/-- Curvature at a fixed time, computed from the spatial slice of Γ. -/
125def curvature (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (t : ℝ) (x X Y Z : E) : E :=
126 fderiv ℝ (Γ t) x X Y Z - fderiv ℝ (Γ t) x Y X Z
127 + Γ t x X (Γ t x Y Z) - Γ t x Y (Γ t x X Z)
128
129/-- Actual explicit time derivative at a fixed chart point. -/
130def timeVariation (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (t : ℝ) (x : E) :=
131 fderiv ℝ (fun s => Γ s x) t 1

133/-- Chain rule for the time-first pullback. Joint differentiability is the only
134coefficient regularity hypothesis; the derivative is not prescribed as input. -/
135theorem fderiv_coefficients {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
136 {u : ℝ × ℝ → E} {p : ℝ × ℝ} (hu : DifferentiableAt ℝ u p)
137 (hΓ : DifferentiableAt ℝ (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p))
138 (d : ℝ × ℝ) :
139 fderiv ℝ (coefficients Γ u) p d =
140 fderiv ℝ (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p) (d.1, fderiv ℝ u p d) := by
141 have h := HasFDerivAt.comp (x := p) (g := fun z : ℝ × E => Γ z.1 z.2)
142 (f := fun q : ℝ × ℝ => (q.1, u q)) hΓ.hasFDerivAt
143 (hasFDerivAt_fst.prodMk hu.hasFDerivAt)
144 rw [show fderiv ℝ (coefficients Γ u) p = _ from h.fderiv]
145 rfl

147/-- The spatial slice derivative is the joint derivative in direction `(0,X)`. -/
148theorem spatial_slice_fderiv {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E} {t : ℝ} {x : E}
149 (hΓ : DifferentiableAt ℝ (fun z : ℝ × E => Γ z.1 z.2) (t, x)) (X : E) :
150 fderiv ℝ (Γ t) x X =
151 fderiv ℝ (fun z : ℝ × E => Γ z.1 z.2) (t, x) (0, X) := by
152 have h := HasFDerivAt.comp (x := x) (g := fun z : ℝ × E => Γ z.1 z.2)
153 (f := fun y : E => (t, y)) hΓ.hasFDerivAt
154 ((hasFDerivAt_const (𝕜 := ℝ) t x).prodMk (hasFDerivAt_id (𝕜 := ℝ) x))
155 rw [show fderiv ℝ (Γ t) x = _ from h.fderiv]
156 rfl

158/-- The time slice derivative is the joint derivative in direction `(1,0)`. -/
159theorem time_slice_fderiv {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E} {t : ℝ} {x : E}
160 (hΓ : DifferentiableAt ℝ (fun z : ℝ × E => Γ z.1 z.2) (t, x)) :
161 timeVariation Γ t x =
162 fderiv ℝ (fun z : ℝ × E => Γ z.1 z.2) (t, x) (1, 0) := by
163 have h := HasFDerivAt.comp (x := t) (g := fun z : ℝ × E => Γ z.1 z.2)
164 (f := fun s : ℝ => (s, x)) hΓ.hasFDerivAt
165 ((hasFDerivAt_id (𝕜 := ℝ) t).prodMk (hasFDerivAt_const (𝕜 := ℝ) x t))
166 unfold timeVariation
167 rw [show fderiv ℝ (fun s => Γ s x) t = _ from h.fderiv]

170/-- Evolving-connection time-space commutator. The additional term is the
171actual time derivative of Γ at fixed position, in addition to spatial curvature. -/
172theorem covTime_covSpace_comm {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
173 {u V : ℝ × ℝ → E} {p : ℝ × ℝ}
174 (hu : ContDiffAt ℝ 2 u p) (hV : ContDiffAt ℝ 2 V p)
175 (hΓ : DifferentiableAt ℝ (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p)) :
176 covTime Γ u (covSpace Γ u V) p - covSpace Γ u (covTime Γ u V) p =
177 curvature Γ p.1 (u p) (fderiv ℝ u p (1, 0)) (fderiv ℝ u p (0, 1)) (V p)
178 + timeVariation Γ p.1 (u p) (fderiv ℝ u p (0, 1)) (V p) := by
179 have hu1 := hu.differentiableAt (by norm_num)
180 have hA : DifferentiableAt ℝ (coefficients Γ u) p :=
181 (HasFDerivAt.comp (x := p) (g := fun z : ℝ × E => Γ z.1 z.2)
182 (f := fun q : ℝ × ℝ => (q.1, u q)) hΓ.hasFDerivAt
183 (hasFDerivAt_fst.prodMk hu1.hasFDerivAt)).differentiableAt
184 change covDerivAlong (coefficients Γ u) u
185 (covDerivAlong (coefficients Γ u) u V (0, 1)) (1, 0) p -
186 covDerivAlong (coefficients Γ u) u
187 (covDerivAlong (coefficients Γ u) u V (1, 0)) (0, 1) p = _
188 rw [covDerivAlong_comm hu hV hA]
189 rw [fderiv_coefficients hu1 hΓ, fderiv_coefficients hu1 hΓ]
190 simp only [coefficients, curvature]
191 rw [spatial_slice_fderiv hΓ, spatial_slice_fderiv hΓ, time_slice_fderiv hΓ]
192 have hp : ((1 : ℝ), fderiv ℝ u p (1, 0)) =
193 (1, 0) + (0, fderiv ℝ u p (1, 0)) := by simp
194 rw [hp, map_add, add_apply, add_apply]
195 abel

197/-- Torsion-free identity for the actual mixed time/spatial velocities. -/
198theorem covTime_spaceVelocity_eq_covSpace_timeVelocity
199 {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E} {u : ℝ × ℝ → E} {p : ℝ × ℝ}
200 (hu : ContDiffAt ℝ 2 u p)
201 (hΓsymm : ∀ X Y, Γ p.1 (u p) X Y = Γ p.1 (u p) Y X) :
202 covTime Γ u (fun q => fderiv ℝ u q (0, 1)) p =
203 covSpace Γ u (fun q => fderiv ℝ u q (1, 0)) p :=
204 covDerivAlong_fderiv_symm hu hΓsymm (1, 0) (0, 1)
