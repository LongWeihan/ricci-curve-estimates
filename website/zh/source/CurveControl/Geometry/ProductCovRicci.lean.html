23/-- **Math.** The actual coordinate covariant derivative of a bilinear tensor. -/
24def covBilinDerivative (Γ : E → E →L[ℝ] E →L[ℝ] E)
25 (A : E → E →L[ℝ] E →L[ℝ] ℝ) (y U V W : E) : ℝ :=
26 fderiv ℝ A y U V W - A y (Γ y U V) W - A y V (Γ y U W)

28/-- **Math.** Ordinary derivative of an actual bilinear coefficient field on a coordinate line. -/
29theorem bilin_line_hasDerivAt {A : E → E →L[ℝ] E →L[ℝ] ℝ} {y : E}
30 (hA : DifferentiableAt ℝ A y) (U V W : E) :
31 HasDerivAt (fun r : ℝ => A (y + r • U) V W) (fderiv ℝ A y U V W) 0 := by
32 have hl : HasDerivAt (fun r : ℝ => y + r • U) U 0 := by
33 simpa only [one_smul, id_eq] using ((hasDerivAt_id (0 : ℝ)).smul_const U).const_add y
34 have hAy : HasFDerivAt A (fderiv ℝ A y) (y + (0 : ℝ) • U) := by
35 simpa only [zero_smul, add_zero] using hA.hasFDerivAt
36 have h := HasFDerivAt.comp_hasDerivAt (𝕜 := ℝ) (l := A)
37 (f := fun r : ℝ => y + r • U) 0 hAy hl
38 have hh := (h.clm_apply (hasDerivAt_const (0 : ℝ) V)).clm_apply (hasDerivAt_const (0 : ℝ) W)
39 simpa using hh

41/-- **Math.** Differentiating the proved splitting of the bilinear coefficients
42under a fixed linear change of model. -/
43theorem fderiv_bilin_split (e : (E × F) ≃L[ℝ] G)
44 {A : E → E →L[ℝ] E →L[ℝ] ℝ} {B : F → F →L[ℝ] F →L[ℝ] ℝ}
45 {C : G → G →L[ℝ] G →L[ℝ] ℝ} {y : G}
46 (hA : DifferentiableAt ℝ A (e.symm y).1)
47 (hB : DifferentiableAt ℝ B (e.symm y).2) (hC : DifferentiableAt ℝ C y)
48 (hsplit : ∀ᶠ z in 𝓝 y, ∀ V W,
49 C z V W = A (e.symm z).1 (e.symm V).1 (e.symm W).1 +
50 B (e.symm z).2 (e.symm V).2 (e.symm W).2) (U V W : G) :
51 fderiv ℝ C y U V W =
52 fderiv ℝ A (e.symm y).1 (e.symm U).1 (e.symm V).1 (e.symm W).1 +
53 fderiv ℝ B (e.symm y).2 (e.symm U).2 (e.symm V).2 (e.symm W).2 := by
54 have hd := bilin_line_hasDerivAt hC U V W
55 have ha := bilin_line_hasDerivAt hA (e.symm U).1 (e.symm V).1 (e.symm W).1
56 have hb := bilin_line_hasDerivAt hB (e.symm U).2 (e.symm V).2 (e.symm W).2
57 have ht : Tendsto (fun r : ℝ => y + r • U) (𝓝 0) (𝓝 y) := by
58 have hct : Continuous (fun r : ℝ => y + r • U) := by fun_prop
59 simpa only [zero_smul, add_zero] using hct.continuousAt.tendsto (x := 0)
60 have heq : (fun r : ℝ => C (y + r • U) V W) =ᶠ[𝓝 0]
61 (fun r => A ((e.symm y).1 + r • (e.symm U).1) (e.symm V).1 (e.symm W).1 +
62 B ((e.symm y).2 + r • (e.symm U).2) (e.symm V).2 (e.symm W).2) := by
63 filter_upwards [ht.eventually hsplit] with r hr
64 simpa only [map_add, map_smul, Prod.fst_add, Prod.snd_add, Prod.smul_fst, Prod.smul_snd] using hr V W
65 exact hd.unique ((ha.add hb).congr_of_eventuallyEq heq)

69theorem covBilinDerivative_split (e : (E × F) ≃L[ℝ] G)
70 {ΓA : E → E →L[ℝ] E →L[ℝ] E} {ΓB : F → F →L[ℝ] F →L[ℝ] F}
