31/-- **Math.** In the chart at the evaluation point the intrinsic curvature
32four-form is minus the chart-curvature metric pairing (upstream convention). -/
33theorem canonicalCurvatureFormAt_eq_chart (g : RiemannianMetric I M)
34 (p : M) (x y z w : TangentSpace I p) :
35 g.leviCivitaConnection.curvatureFormAt g p x y z w =
36 -chartMetricInner g p (extChartAt I p p)
37 (chartCurvature g p (extChartAt I p p) x y z) w := by
38 have hh := curvatureFormAt_chartFrame g (α := p) (p := p) (mem_chart_source H p) x y z w
39 rw [← trivializationAt_symm_eq_sum_chartBasisVecFiber (I := I) p p x (mem_chart_source H p),
40 ← trivializationAt_symm_eq_sum_chartBasisVecFiber (I := I) p p y (mem_chart_source H p),
41 ← trivializationAt_symm_eq_sum_chartBasisVecFiber (I := I) p p z (mem_chart_source H p),
42 ← trivializationAt_symm_eq_sum_chartBasisVecFiber (I := I) p p w (mem_chart_source H p),
43 trivializationAt_symm_self (I := I), trivializationAt_symm_self (I := I),
44 trivializationAt_symm_self (I := I), trivializationAt_symm_self (I := I)] at hh
45 exact hh

111/-- **Math.** The canonical intrinsic curvature four-form of the actual L2
112product metric is the sum of the two factor curvature four-forms. -/
113theorem canonicalCurvatureFormAt_productL2
114 (g : RiemannianMetric I M) (h : RiemannianMetric J N)
115 (p : M × N) (x y z w : TangentSpace (productL2Model I J) p) :
116 (productL2Metric I J g h).leviCivitaConnection.curvatureFormAt (productL2Metric I J g h)
117 p x y z w =

119 (productL2Equiv.symm x).1 (productL2Equiv.symm y).1
120 (productL2Equiv.symm z).1 (productL2Equiv.symm w).1 +
121 h.leviCivitaConnection.curvatureFormAt h p.2
122 (productL2Equiv.symm x).2 (productL2Equiv.symm y).2
123 (productL2Equiv.symm z).2 (productL2Equiv.symm w).2 := by
124 have hcoords : productL2Equiv.symm (extChartAt (productL2Model I J) p p) =
