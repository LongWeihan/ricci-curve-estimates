313/-- **Math.** Curvature-vector evolution in an actual fixed manifold chart.
314Every datum is derived from the genuine curve flow and Ricci flow. In particular,
315the speed equation, chart normal identity, torsion freedom and coefficient
316regularity are conclusions of the preceding geometric lemmas, not new inputs. -/
317theorem normal_evolution_in_chart [SigmaCompactSpace M] [T2Space M]
318 (hflow : IsRicciFlowOn g J) (α : M) {p : ℝ × ℝ}
319 (hp : p.1 ∈ interior J) (hα : c p.1 p.2 ∈ (chartAt H α).source) :

323 let S := CurveEvolution.tangent u v
324 let N := CurveEvolution.normal Γ u v
325 EvolvingConnection.covTime Γ u N p =
326 CurveEvolution.covArc Γ u v (CurveEvolution.covArc Γ u v N) p
327 + (2 * curveSpeedDecay g c p) • N p
328 + CurveEvolution.scalarArc v (curveSpeedDecay g c) p • S p
329 + EvolvingConnection.curvature Γ p.1 (u p) (N p) (S p) (S p)
330 + EvolvingConnection.timeVariation Γ p.1 (u p) (S p) (S p) := by
331 let Γ := fun t => chartChristoffelBilin (I := I) (g t) α
332 let u := chartCurveFamily (I := I) α c
333 let v := fun z : ℝ × ℝ => curveSpeed (g z.1) (c z.1) z.2
334 have hu : ContDiffAt ℝ 4 u p :=
335 contDiffAt_infty.mp (hc.contDiffAt_chartCurveFamily α hp hα) 4
336 have hv : ContDiffAt ℝ 3 v p :=
337 contDiffAt_infty.mp (hc.contDiffAt_curveSpeed hflow.smooth hp) 3
