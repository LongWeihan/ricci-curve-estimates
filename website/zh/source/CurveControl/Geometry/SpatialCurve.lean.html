19/-- A smooth curve has a smooth representative in its moving-foot chart. -/
20theorem contDiffAt_chartLocalCurve (γ : ℝ → M) (x : ℝ)
21 (hγ : ContMDiffAt 𝓘(ℝ, ℝ) I ∞ γ x) :
22 ContDiffAt ℝ ∞ (chartLocalCurve (I := I) γ x) x := by
23 exact (contMDiffAt_extChartAt.comp x hγ).contDiffAt

25/-- Continuity keeps the curve in the anchor chart near its base parameter. -/
26theorem eventually_mem_chartSource (γ : ℝ → M) (x : ℝ)
27 (hγ : ContinuousAt γ x) :
28 ∀ᶠ y in 𝓝 x, γ y ∈ (chartAt H (γ x)).source :=
29 hγ ((chartAt H (γ x)).open_source.mem_nhds (mem_chart_source H (γ x)))

31/-- Differentiability of the actual chart representatives suffices to eliminate
32the totalized-derivative junk in `covDerivAlong`. -/
33theorem hasCovDerivAlongAt_covDerivAlong (g : RiemannianMetric I M)
34 (γ : ℝ → M) (W : ∀ x, TangentSpace I (γ x)) (x : ℝ)
35 (hmem : ∀ᶠ y in 𝓝 x, γ y ∈ (chartAt H (γ x)).source)
36 (hγ : DifferentiableAt ℝ (chartLocalCurve (I := I) γ x) x)
37 (hW : DifferentiableAt ℝ (chartFieldCoord (I := I) (γ x) γ W) x) :
38 HasCovDerivAlongAt (I := I) g γ W x (covDerivAlong g γ W x) := by
39 exact ⟨hmem, _, _, hγ.hasDerivAt, hW.hasDerivAt, rfl⟩

41/-- The velocity field has smooth coordinates near every parameter. This uses
42the chart-transition theorem, rather than differentiating moving charts. -/
43theorem contDiffAt_chartFieldCoord_velocity (γ : ℝ → M)
44 (hγ : ContMDiff 𝓘(ℝ, ℝ) I ∞ γ) (x : ℝ) :
45 ContDiffAt ℝ ∞
46 (chartFieldCoord (I := I) (γ x) γ (curveVelocity (I := I) γ)) x := by
47 have hc := contDiffAt_chartLocalCurve γ x (hγ.contMDiffAt)
48 have hd : ContDiffAt ℝ ∞ (deriv (chartLocalCurve (I := I) γ x)) x :=
49 hc.derivWithin (by simp)
50 apply hd.congr_of_eventuallyEq
51 filter_upwards [eventually_mem_chartSource (H := H) γ x (hγ.continuous.continuousAt)] with y hy
52 have hmem : ∀ᶠ z in 𝓝 y, γ z ∈ (chartAt H (γ x)).source :=
53 hγ.continuous.continuousAt ((chartAt H (γ x)).open_source.mem_nhds hy)
54 have hchart : ContDiffAt ℝ ∞ (chartLocalCurve (I := I) γ x) y :=
55 ((contMDiffAt_extChartAt' hy).comp y hγ.contMDiffAt).contDiffAt
56 exact chartFieldCoord_curveVelocity_eq hmem
57 (hchart.differentiableAt (by simp)).hasDerivAt

59/-- Smoothness of the curve supplies the actual covariant acceleration. -/
60theorem hasCovDerivAlongAt_velocity (g : RiemannianMetric I M) (γ : ℝ → M)
61 (hγ : ContMDiff 𝓘(ℝ, ℝ) I ∞ γ) (x : ℝ) :
62 HasCovDerivAlongAt (I := I) g γ (curveVelocity (I := I) γ) x
63 (covDerivAlong g γ (curveVelocity (I := I) γ) x) := by
64 apply hasCovDerivAlongAt_covDerivAlong
65 · exact eventually_mem_chartSource (H := H) γ x hγ.continuous.continuousAt
66 · exact (contDiffAt_chartLocalCurve γ x hγ.contMDiffAt).differentiableAt (by simp)
67 · exact (contDiffAt_chartFieldCoord_velocity γ hγ x).differentiableAt (by simp)

69/-- Spatial speed derivative in a fixed ambient metric, derived from metric
70compatibility. The covariant derivative hypothesis is the upstream genuine
71chart differentiability predicate, not a speed evolution assumption. -/
72theorem hasDerivAt_curveSpeed (g : RiemannianMetric I M) (γ : ℝ → M)
73 (x : ℝ) (D : E) (himm : curveVelocity (I := I) γ x ≠ 0)
74 (hD : HasCovDerivAlongAt (I := I) g γ (curveVelocity (I := I) γ) x D) :
75 HasDerivAt (curveSpeed g γ)
76 ((g.metricInner (γ x) D (curveVelocity (I := I) γ x) +
77 g.metricInner (γ x) (curveVelocity (I := I) γ x) D) /
78 (2 * curveSpeed g γ x)) x := by
79 exact (hD.hasDerivAt_metricInner hD).sqrt
80 (ne_of_gt (g.metricInner_self_pos _ _ himm))

82/-- Curvature is orthogonal to the unit tangent. Unit length is proved from
83immersion, and metric compatibility differentiates that identity. -/
84theorem curvatureVector_orthogonal (g : RiemannianMetric I M) (γ : ℝ → M)
85 (x : ℝ) (himm : ∀ y, curveVelocity (I := I) γ y ≠ 0)
86 (hS : HasCovDerivAlongAt (I := I) g γ (unitTangent g γ) x
87 (covDerivAlong g γ (unitTangent g γ) x)) :
88 g.metricInner (γ x) (curvatureVector g γ x) (unitTangent g γ x) = 0 := by
89 have hconst : (fun y => g.metricInner (γ y) (unitTangent g γ y)
90 (unitTangent g γ y)) = fun _ => (1 : ℝ) := by
91 funext y
92 exact unitTangent_unit g γ y (himm y)
93 have hprod := hS.hasDerivAt_metricInner hS
94 rw [hconst] at hprod
95 have hzero := hprod.unique (hasDerivAt_const x (1 : ℝ))
96 rw [g.metricInner_comm (γ x) (unitTangent g γ x)
97 (covDerivAlong g γ (unitTangent g γ) x)] at hzero
98 have hperp : g.metricInner (γ x) (covDerivAlong g γ (unitTangent g γ) x)
99 (unitTangent g γ x) = 0 := by linarith
100 unfold curvatureVector
101 rw [g.metricInner_smul_left, hperp, mul_zero]

103/-- Unit tangent covariant differentiability follows from smooth immersion. -/
104theorem hasCovDerivAlongAt_unitTangent (g : RiemannianMetric I M) (γ : ℝ → M)
105 (hγ : ContMDiff 𝓘(ℝ, ℝ) I ∞ γ) (x : ℝ)
106 (himm : curveVelocity (I := I) γ x ≠ 0) :
107 HasCovDerivAlongAt (I := I) g γ (unitTangent g γ) x
108 (covDerivAlong g γ (unitTangent g γ) x) := by
109 have hX := hasCovDerivAlongAt_velocity g γ hγ x
110 have hv := hasDerivAt_curveSpeed g γ x _ himm hX
111 have hinv := hv.inv (ne_of_gt (curveSpeed_pos g γ x himm))
112 have hS := HasCovDerivAlongAt.smul_fun hinv hX
113 change HasCovDerivAlongAt (I := I) g γ (unitTangent g γ) x _ at hS
114 rw [covDerivAlong_eq_of_hasCovDeriv g γ (unitTangent g γ) x _ hS]
115 exact hS

117/-- Orthogonality for a smooth immersed curve, with no derivative or
118orthogonality hypothesis on its curvature vector. -/
119theorem curvatureVector_orthogonal_of_contMDiff (g : RiemannianMetric I M)
120 (γ : ℝ → M) (hγ : ContMDiff 𝓘(ℝ, ℝ) I ∞ γ)
121 (himm : ∀ y, curveVelocity (I := I) γ y ≠ 0) (x : ℝ) :
122 g.metricInner (γ x) (curvatureVector g γ x) (unitTangent g γ x) = 0 :=
123 curvatureVector_orthogonal g γ x himm
124 (hasCovDerivAlongAt_unitTangent g γ hγ x (himm x))
