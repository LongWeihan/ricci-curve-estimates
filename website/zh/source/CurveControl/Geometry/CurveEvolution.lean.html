4/-!
5# Coordinate evolution of tangent and curvature
6
7All vector fields below are defined from actual Fréchet derivatives. The only
8geometric evolution inputs are curve shortening and the independent speed
9identity. Connection commutation is the proved time-dependent identity.
10-/
11open Filter
12open scoped Topology ContDiff

19/-- Arc-length covariant differentiation. -/
20def covArc (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (u : ℝ × ℝ → E)
21 (v : ℝ × ℝ → ℝ) (V : ℝ × ℝ → E) (p : ℝ × ℝ) : E :=
22 (v p)⁻¹ • covSpace Γ u V p
23
24/-- Actual unit tangent when v is the metric speed. -/
25def tangent (u : ℝ × ℝ → E) (v : ℝ × ℝ → ℝ) (p : ℝ × ℝ) : E :=
26 (v p)⁻¹ • fderiv ℝ u p (0, 1)
27
28/-- Actual curvature vector for arc-length differentiation. -/
29def normal (Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E) (u : ℝ × ℝ → E)
30 (v : ℝ × ℝ → ℝ) := covArc Γ u v (tangent u v)
31
32def scalarArc (v a : ℝ × ℝ → ℝ) (p : ℝ × ℝ) : ℝ :=
33 (v p)⁻¹ * fderiv ℝ a p (0, 1)

35theorem covDerivAlong_congr {A : P → E →L[ℝ] E →L[ℝ] E} {u V W : P → E}
36 {p : P} (h : V =ᶠ[𝓝 p] W) (d : P) :
37 covDerivAlong A u V d p = covDerivAlong A u W d p := by
38 simp only [covDerivAlong, h.fderiv_eq, h.eq_of_nhds]

40theorem covDerivAlong_smul {A : P → E →L[ℝ] E →L[ℝ] E} {u V : P → E}
41 {f : P → ℝ} {p : P} (hf : DifferentiableAt ℝ f p)
42 (hV : DifferentiableAt ℝ V p) (d : P) :
43 covDerivAlong A u (fun q => f q • V q) d p =
44 (fderiv ℝ f p d) • V p + f p • covDerivAlong A u V d p := by
45 simp only [covDerivAlong, fderiv_fun_smul hf hV, add_apply,
46 smul_apply, ContinuousLinearMap.smulRight_apply,
47 map_smul, smul_add]

50theorem covDerivAlong_add {A : P → E →L[ℝ] E →L[ℝ] E} {u V W : P → E}
51 {p : P} (hV : DifferentiableAt ℝ V p) (hW : DifferentiableAt ℝ W p) (d : P) :
52 covDerivAlong A u (fun q => V q + W q) d p =
53 covDerivAlong A u V d p + covDerivAlong A u W d p := by
54 simp only [covDerivAlong, fderiv_fun_add hV hW, add_apply, map_add]

57/-- Inverse-speed derivative follows from the actual scalar chain rule. -/
58theorem inverse_speed_time {v : ℝ × ℝ → ℝ} {p : ℝ × ℝ} {a : ℝ}
59 (hv : DifferentiableAt ℝ v p) (hvp : v p ≠ 0)
60 (hevol : fderiv ℝ v p (1, 0) = -a * v p) :
61 fderiv ℝ (fun q => (v q)⁻¹) p (1, 0) = a * (v p)⁻¹ := by
62 have h := (hasDerivAt_inv hvp).comp_hasFDerivAt p hv.hasFDerivAt
63 rw [show fderiv ℝ (fun q => (v q)⁻¹) p = _ from h.fderiv]
64 simp only [smul_apply, smul_eq_mul, hevol]
65 field_simp

67/-- Evolution of S, using Schwarz symmetry and torsion-freeness, not an
68assumed evolution equation for S. The curve-shortening identity holds as a germ. -/
69theorem tangent_evolution {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
70 {u : ℝ × ℝ → E} {v : ℝ × ℝ → ℝ} {p : ℝ × ℝ} {a : ℝ}
71 (hu : ContDiffAt ℝ 2 u p) (hv : DifferentiableAt ℝ v p) (hvp : v p ≠ 0)
72 (hΓsymm : ∀ X Y, Γ p.1 (u p) X Y = Γ p.1 (u p) Y X)
73 (hflow : (fun q => fderiv ℝ u q (1, 0)) =ᶠ[𝓝 p] normal Γ u v)
74 (hspeed : fderiv ℝ v p (1, 0) = -a * v p) :
75 covTime Γ u (tangent u v) p = covArc Γ u v (normal Γ u v) p + a • tangent u v p := by
76 have hux : DifferentiableAt ℝ (fun q => fderiv ℝ u q (0, 1)) p :=
77 ((hu.fderiv_right (m := 1) (by norm_num)).clm_apply contDiffAt_const).differentiableAt
78 (by norm_num)
79 change covDerivAlong (coefficients Γ u) u
80 (fun q => (v q)⁻¹ • fderiv ℝ u q (0, 1)) (1, 0) p = _
81 rw [covDerivAlong_smul (f := fun q => (v q)⁻¹) (hv.inv hvp) hux, inverse_speed_time hv hvp hspeed]
82 rw [covDerivAlong_fderiv_symm hu hΓsymm (1, 0) (0, 1)]
83 rw [covDerivAlong_congr hflow (0, 1)]
84 simp only [covArc, tangent, covSpace, mul_smul]
85 abel

87/-- Smoothness of the actual normalized tangent is derived from u and v. -/
88theorem tangent_contDiffAt {u : ℝ × ℝ → E} {v : ℝ × ℝ → ℝ} {p : ℝ × ℝ}
89 (hu : ContDiffAt ℝ 4 u p) (hv : ContDiffAt ℝ 3 v p) (hvp : v p ≠ 0) :
90 ContDiffAt ℝ 3 (tangent u v) p :=
91 (hv.inv hvp).smul ((hu.fderiv_right (m := 3) (by norm_num)).clm_apply contDiffAt_const)

93/-- Regularity of the actual coefficient pullback. -/
94theorem coefficients_contDiffAt {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
95 {u : ℝ × ℝ → E} {p : ℝ × ℝ}
96 (hu : ContDiffAt ℝ 2 u p)
97 (hΓ : ContDiffAt ℝ 2 (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p)) :
98 ContDiffAt ℝ 2 (coefficients Γ u) p :=
99 ContDiffAt.comp (f := fun q : ℝ × ℝ => (q.1, u q))
100 (g := fun z : ℝ × E => Γ z.1 z.2) p hΓ (contDiffAt_fst.prodMk hu)

102/-- Regularity for differentiating the spatial covariant derivative once more. -/
103theorem covSpace_contDiffAt {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
104 {u V : ℝ × ℝ → E} {p : ℝ × ℝ}
105 (hu : ContDiffAt ℝ 3 u p) (hV : ContDiffAt ℝ 3 V p)
106 (hΓ : ContDiffAt ℝ 2 (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p)) :
107 ContDiffAt ℝ 2 (covSpace Γ u V) p := by
108 have hA := coefficients_contDiffAt (hu.of_le (by norm_num)) hΓ
109 exact ((hV.fderiv_right (m := 2) (by norm_num)).clm_apply contDiffAt_const).add
110 ((hA.clm_apply ((hu.fderiv_right (m := 2) (by norm_num)).clm_apply contDiffAt_const)).clm_apply
111 (hV.of_le (by norm_num)))

113/-- Curvature-vector C² regularity is derived, not supplied as geometric data. -/
114theorem normal_contDiffAt {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
115 {u : ℝ × ℝ → E} {v : ℝ × ℝ → ℝ} {p : ℝ × ℝ}
116 (hu : ContDiffAt ℝ 4 u p) (hv : ContDiffAt ℝ 3 v p) (hvp : v p ≠ 0)
117 (hΓ : ContDiffAt ℝ 2 (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p)) :
118 ContDiffAt ℝ 2 (normal Γ u v) p :=
119 ((hv.of_le (by norm_num)).inv hvp).smul
120 (covSpace_contDiffAt (hu.of_le (by norm_num)) (tangent_contDiffAt hu hv hvp) hΓ)

124 curvature Γ t x X (c • Y) Z = c • curvature Γ t x X Y Z := by

127/-- Commuting time with the genuine arc-length derivative includes the speed
128variation term and the explicit time derivative of the spatial connection. -/
129theorem covTime_covArc_comm {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
130 {u V : ℝ × ℝ → E} {v : ℝ × ℝ → ℝ} {p : ℝ × ℝ} {a : ℝ}
131 (hu : ContDiffAt ℝ 3 u p) (hV : ContDiffAt ℝ 3 V p)
132 (hv : DifferentiableAt ℝ v p) (hvp : v p ≠ 0)
133 (hΓ : ContDiffAt ℝ 2 (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p))
134 (hspeed : fderiv ℝ v p (1, 0) = -a * v p) :
135 covTime Γ u (covArc Γ u v V) p = covArc Γ u v (covTime Γ u V) p
136 + a • covArc Γ u v V p
137 + curvature Γ p.1 (u p) (fderiv ℝ u p (1, 0)) (tangent u v p) (V p)
138 + timeVariation Γ p.1 (u p) (tangent u v p) (V p) := by
139 have hdx := (covSpace_contDiffAt hu hV hΓ).differentiableAt (by norm_num)
140 change covDerivAlong (coefficients Γ u) u
141 (fun q => (v q)⁻¹ • covSpace Γ u V q) (1, 0) p = _
142 rw [covDerivAlong_smul (f := fun q => (v q)⁻¹) (hv.inv hvp) hdx,
143 inverse_speed_time hv hvp hspeed]
144 have hc := covTime_covSpace_comm (hu.of_le (by norm_num)) (hV.of_le (by norm_num))
145 (hΓ.differentiableAt (by norm_num))
146 have he := eq_add_of_sub_eq hc
147 change covTime Γ u (covSpace Γ u V) p = _ at he
148 change (a * (v p)⁻¹) • covSpace Γ u V p +
149 (v p)⁻¹ • covTime Γ u (covSpace Γ u V) p = _
150 rw [he]
151 simp only [covArc, tangent, curvature_smul_second, map_smul, smul_apply, smul_add, mul_smul]
152 abel

154/-- C² vector fields have differentiable spatial covariant derivative. -/
155theorem covSpace_differentiableAt {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
156 {u V : ℝ × ℝ → E} {p : ℝ × ℝ}
157 (hu : ContDiffAt ℝ 2 u p) (hV : ContDiffAt ℝ 2 V p)
158 (hΓ : ContDiffAt ℝ 2 (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p)) :
159 DifferentiableAt ℝ (covSpace Γ u V) p := by
160 have hA := (coefficients_contDiffAt hu hΓ).differentiableAt (by norm_num)
161 have hdu := ((hu.fderiv_right (m := 1) (by norm_num)).clm_apply
162 (contDiffAt_const (c := (0, 1)))).differentiableAt (by norm_num)
163 have hdV := ((hV.fderiv_right (m := 1) (by norm_num)).clm_apply
164 (contDiffAt_const (c := (0, 1)))).differentiableAt (by norm_num)
165 exact hdV.add ((hA.clm_apply hdu).clm_apply (hV.differentiableAt (by norm_num)))

167/-- Local curvature-vector evolution. All regularity of S and H is obtained
168from the displayed smooth data; the two germ evolution inputs concern only
169u and speed v. This identity retains the explicit evolving-connection term. -/
170theorem normal_evolution {Γ : ℝ → E → E →L[ℝ] E →L[ℝ] E}
171 {u : ℝ × ℝ → E} {v a : ℝ × ℝ → ℝ} {p : ℝ × ℝ}
172 (hu : ContDiffAt ℝ 4 u p) (hv : ContDiffAt ℝ 3 v p) (hvp : v p ≠ 0)
173 (ha : DifferentiableAt ℝ a p)
174 (hΓ : ContDiffAt ℝ 2 (fun z : ℝ × E => Γ z.1 z.2) (p.1, u p))
175 (hΓsymm : ∀ᶠ q in 𝓝 p, ∀ X Y, Γ q.1 (u q) X Y = Γ q.1 (u q) Y X)
176 (hflow : (fun q => fderiv ℝ u q (1, 0)) =ᶠ[𝓝 p] normal Γ u v)
177 (hspeed : ∀ᶠ q in 𝓝 p, fderiv ℝ v q (1, 0) = -a q * v q) :
178 covTime Γ u (normal Γ u v) p =
179 covArc Γ u v (covArc Γ u v (normal Γ u v)) p
180 + (2 * a p) • normal Γ u v p
181 + scalarArc v a p • tangent u v p
182 + curvature Γ p.1 (u p) (normal Γ u v p) (tangent u v p) (tangent u v p)
183 + timeVariation Γ p.1 (u p) (tangent u v p) (tangent u v p) := by
184 have hu2 : ContDiffAt ℝ 2 u p := hu.of_le (by norm_num)
185 have hu3 : ContDiffAt ℝ 3 u p := hu.of_le (by norm_num)
186 have hv1 := hv.differentiableAt (by norm_num)
187 have hS3 := tangent_contDiffAt hu hv hvp
188 have hH2 := normal_contDiffAt hu hv hvp hΓ
189 have hS1 := hS3.differentiableAt (by norm_num)
190 have hDsH : DifferentiableAt ℝ (covArc Γ u v (normal Γ u v)) p :=
191 (hv1.inv hvp).smul (covSpace_differentiableAt hu2 hH2 hΓ)
192 have hnear : covTime Γ u (tangent u v) =ᶠ[𝓝 p]
193 (fun q => covArc Γ u v (normal Γ u v) q + a q • tangent u v q) := by
194 have hureg := hu.eventually (by norm_num)
195 have hvreg := hv.eventually (by norm_num)
196 have hvnz := hv.continuousAt.eventually_ne hvp
197 filter_upwards [hureg, hvreg, hvnz, hΓsymm, hflow.eventually_nhds, hspeed]
198 with q huq hvq hvnq hΓq hflowq hspeedq
199 exact tangent_evolution (huq.of_le (by norm_num))
200 (hvq.differentiableAt (by norm_num)) hvnq hΓq hflowq hspeedq
201 have hcomm := covTime_covArc_comm hu3 hS3 hv1 hvp hΓ (hspeed.self_of_nhds)
202 change covTime Γ u (normal Γ u v) p = _ at hcomm
203 rw [hflow.eq_of_nhds] at hcomm
204 rw [hcomm]
205 have hrewrite : covArc Γ u v (covTime Γ u (tangent u v)) p =
206 covArc Γ u v (covArc Γ u v (normal Γ u v)) p
207 + scalarArc v a p • tangent u v p + a p • normal Γ u v p := by
208 change (v p)⁻¹ • covDerivAlong (coefficients Γ u) u
209 (covTime Γ u (tangent u v)) (0, 1) p = _
210 rw [covDerivAlong_congr hnear (0, 1),
211 covDerivAlong_add (W := fun q => a q • tangent u v q) hDsH (ha.smul hS1),
212 covDerivAlong_smul ha hS1]
213 change (v p)⁻¹ • (covDerivAlong (coefficients Γ u) u
214 (covArc Γ u v (normal Γ u v)) (0, 1) p +
215 (fderiv ℝ a p (0, 1) • tangent u v p +
216 a p • covDerivAlong (coefficients Γ u) u (tangent u v) (0, 1) p)) =
217 (v p)⁻¹ • covDerivAlong (coefficients Γ u) u
218 (covArc Γ u v (normal Γ u v)) (0, 1) p +
219 ((v p)⁻¹ * fderiv ℝ a p (0, 1)) • tangent u v p +
220 a p • ((v p)⁻¹ • covDerivAlong (coefficients Γ u) u (tangent u v) (0, 1) p)
221 simp only [smul_add, mul_smul]
222 rw [smul_comm (v p)⁻¹ (a p)]
223 abel
224 rw [hrewrite]
225 change _ + a p • normal Γ u v p + _ + _ = _
226 simp only [show (2 : ℝ) * a p = a p + a p by ring, add_smul]
227 abel
