29/-- **Math.** The corrected ambient contribution admits one coefficient
30multiplying q plus square root q; that coefficient is independent of the curve. -/
31theorem curve_ambient_error_le_max
32 {g : Riemannian.RiemannianMetric I M} (γ : ℝ → M) (x : ℝ)
33 {B K D : ℝ} (hBounds : AmbientBounds.TensorBoundsAt g (γ x) B K D)
34 (himm : curveVelocity (I := I) γ x ≠ 0) :
35 let S := unitTangent g γ x
36 let V := curvatureVector g γ x
37 ((4 * curvatureSq g γ x * ricciTensorAt g (γ x) S S -
38 2 * ricciTensorAt g (γ x) V V + 2 * AmbientBounds.rm g (γ x) V S V S) +
39 (-4 * AmbientBounds.covRic g (γ x) S S V + 2 * AmbientBounds.covRic g (γ x) V S S)) ≤
40 max (6 * B + 2 * K) (6 * D) *
41 (curvatureSq g γ x + Real.sqrt (curvatureSq g γ x)) := by
42 dsimp only
43 have h := (le_abs_self _).trans (AmbientBounds.curve_ambient_error γ x hBounds himm)
44 have hq := mul_le_mul_of_nonneg_right (le_max_left (6 * B + 2 * K) (6 * D))
45 (curvatureSq_nonneg g γ x)
46 have hk := mul_le_mul_of_nonneg_right (le_max_right (6 * B + 2 * K) (6 * D))
47 (curvature_nonneg g γ x)
48 change _ ≤ max (6 * B + 2 * K) (6 * D) * (curvatureSq g γ x + curvature g γ x)
49 nlinarith

114/-- **Math.** The actual squared-curvature PDE with a uniform ambient
115coefficient. Its evolution, normal-gradient dissipation, and tensor error all
116come from proved geometric identities. -/
117theorem curvatureSq_time_le {B K D : ℝ}
118 (hBounds : AmbientBounds.UniformTensorBoundsOn g J B K D)
119 {t : ℝ} (ht : t ∈ interior J) (x : ℝ) :
120 deriv (fun u => curvatureSq (g u) (c u) x) t ≤
121 Analysis.arcDeriv (curveSpeed (g t) (c t))
122 (Analysis.arcDeriv (curveSpeed (g t) (c t)) (curvatureSq (g t) (c t))) x -
123 2 * normalCurvatureGradientNorm (g t) (c t) x ^ 2 +
124 2 * curvatureSq (g t) (c t) x ^ 2 +
125 max (6 * B + 2 * K) (6 * D) *
126 (curvatureSq (g t) (c t) x + Real.sqrt (curvatureSq (g t) (c t) x)) := by
127 have hq := (hc.hasDerivAt_curvatureSq_time hflow ht x).deriv
128 have herr := curve_ambient_error_le_max (c t) x
129 (hBounds t (interior_subset ht) (c t x)) (hc.immersed t (interior_subset ht) x)
130 dsimp only at hq herr
131 rw [hq]
132 dsimp only [normalCurvatureGradientNorm]
133 rw [Real.sq_sqrt ((g t).metricInner_self_nonneg _ _)]

136/-- **Math.** The complete actual length and total-curvature estimate. The
137same ambient constants work for every immersed closed curve-shortening flow.
138Only smooth existence, the Ricci/curve-flow equations and ambient tensor bounds
139are inputs; there are no curvature-PDE, Kato, regularization or integral inputs. -/
140theorem curve_estimate {a b B K D : ℝ} (hab : a ≤ b) (hJ : Icc a b ⊆ J)
141 (hBounds : AmbientBounds.UniformTensorBoundsOn g J B K D) :
142 ∀ t ∈ Icc a b,
143 curveLength (g t) (c t) ≤ curveLength (g a) (c a) * Real.exp (B * (t - a)) ∧
144 totalCurvature (g t) (c t) + curveLength (g t) (c t) ≤
145 (totalCurvature (g a) (c a) + curveLength (g a) (c a)) *
146 Real.exp ((max (6 * B + 2 * K) (6 * D) / 2 + B) * (t - a)) := by
147 have hJi : Ioo a b ⊆ interior J := by
148 intro t ht
149 exact mem_interior_iff_mem_nhds.mpr (Filter.mem_of_superset (Icc_mem_nhds ht.1 ht.2) hJ)
150 have hs := hc.scalarCurveHypotheses_of_squaredPDE hflow hab hJ hBounds
151 (fun t ht x => hc.curvatureSq_time_le hflow hBounds (hJi ht) x)
152 simpa only [CurveControl.weightedCurvature, totalCurvature, curveLength, curvature] using
153 hs.curve_estimate

170/-- **Math.** Uniform initial length and total-curvature bounds propagate
171uniformly across the family with coefficients depending only on the ambient
172metric bounds. -/
173theorem curve_family_uniform_estimate {ι : Type*}
174 {g : ℝ → Riemannian.RiemannianMetric I M} {c : ι → ℝ → ℝ → M} {J : Set ℝ}
175 (hc : ∀ i, IsCurveShorteningFlowOn (I := I) g (c i) J) (hflow : IsRicciFlowOn g J)
176 {a b B K D L₀ Q₀ : ℝ} (hab : a ≤ b) (hJ : Icc a b ⊆ J)
177 (hBounds : AmbientBounds.UniformTensorBoundsOn g J B K D)
178 (hL₀ : ∀ i, curveLength (g a) (c i a) ≤ L₀)
179 (hQ₀ : ∀ i, totalCurvature (g a) (c i a) + curveLength (g a) (c i a) ≤ Q₀) :
180 ∀ i, ∀ t ∈ Icc a b,
181 curveLength (g t) (c i t) ≤ L₀ * Real.exp (B * (t - a)) ∧
182 totalCurvature (g t) (c i t) + curveLength (g t) (c i t) ≤
183 Q₀ * Real.exp ((max (6 * B + 2 * K) (6 * D) / 2 + B) * (t - a)) := by
184 intro i t ht
185 have h := (hc i).curve_estimate hflow hab hJ hBounds t ht
186 exact ⟨h.1.trans (mul_le_mul_of_nonneg_right (hL₀ i) (Real.exp_pos _).le),
187 h.2.trans (mul_le_mul_of_nonneg_right (hQ₀ i) (Real.exp_pos _).le)⟩
