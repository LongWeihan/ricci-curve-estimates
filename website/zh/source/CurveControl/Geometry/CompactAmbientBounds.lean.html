34/-- **Math.** A weighted finite sum bound, used one tensor slot at a time. -/
35theorem abs_sum_mul_le {ι : Type*} [Fintype ι] (a f : ι → ℝ) (C : ℝ)
36 (hf : ∀ i, |f i| ≤ C) :
37 |∑ i, a i * f i| ≤ (∑ i, |a i|) * C := by
38 calc
39 _ ≤ ∑ i, |a i * f i| := Finset.abs_sum_le_sum_abs _ _
40 _ ≤ ∑ i, |a i| * C := Finset.sum_le_sum fun i _ => by
41 rw [abs_mul]
42 exact mul_le_mul_of_nonneg_left (hf i) (abs_nonneg _)
43 _ = _ := (Finset.sum_mul _ _ _).symm

46theorem abs_sum_mul₂_le {ι : Type*} [Fintype ι]
47 (a b : ι → ℝ) (T : ι → ι → ℝ) (C : ℝ) (hT : ∀ i j, |T i j| ≤ C) :
48 |∑ i, ∑ j, a i * b j * T i j| ≤ (∑ i, |a i|) * ((∑ j, |b j|) * C) := by
49 have h := abs_sum_mul_le a (fun i => ∑ j, b j * T i j)
