6# Controlling chart coordinates by the metric norm

8The constant is the square root of an actual inverse Gram entry. This bridges
9compact bounds on tensor coefficients to intrinsic bounds on tangent vectors.

23/-- **Math.** The metric dual of a chart coordinate functional. -/
24def chartMetricDual (g : RiemannianMetric I M) (α : M) {p : M}
25 (hp : p ∈ (trivializationAt E (TangentSpace I) α).baseSet)
26 (i : Fin (Module.finrank ℝ E)) : TangentSpace I p :=
27 ∑ j, chartInvGramMatrix g α p i j • chartBasisFamily α hp j

29theorem chartMetricDual_pairing_basis (g : RiemannianMetric I M) (α : M) {p : M}
30 (hp : p ∈ (trivializationAt E (TangentSpace I) α).baseSet)
31 (i k : Fin (Module.finrank ℝ E)) :
32 g.metricInner p (chartMetricDual g α hp i) (chartBasisFamily α hp k) =
33 if i = k then 1 else 0 := by
34 classical
35 change g.inner p (∑ j, chartInvGramMatrix g α p i j • chartBasisFamily α hp j)
36 (chartBasisFamily α hp k) = _
37 simp only [map_sum, sum_apply, map_smul,
38 smul_apply, smul_eq_mul, chartBasisFamily_apply]
39 have h := congrArg (fun A => A i k) (chartInvGramMatrix_mul_chartGramMatrix g α hp)
40 simpa only [Matrix.mul_apply, Matrix.one_apply, chartGramMatrix_apply] using h
41
42theorem chartMetricDual_pairing (g : RiemannianMetric I M) (α : M) {p : M}
43 (hp : p ∈ (trivializationAt E (TangentSpace I) α).baseSet)
44 (i : Fin (Module.finrank ℝ E)) (v : TangentSpace I p) :
45 g.metricInner p (chartMetricDual g α hp i) v = (chartBasisFamily α hp).repr v i := by
46 classical
47 let b := chartBasisFamily (I := I) α hp
48 calc
49 _ = g.inner p (chartMetricDual g α hp i) (∑ j, b.repr v j • b j) := by
50 rw [b.sum_repr]
51 rfl
52 _ = ∑ j, b.repr v j * (if i = j then 1 else 0) := by
53 simp only [map_sum, map_smul, smul_eq_mul]
54 apply Finset.sum_congr rfl
55 intro j _
56 congr 1
57 exact chartMetricDual_pairing_basis g α hp i j
58 _ = b.repr v i := by simp

60theorem chartMetricDual_sq (g : RiemannianMetric I M) (α : M) {p : M}
61 (hp : p ∈ (trivializationAt E (TangentSpace I) α).baseSet)
62 (i : Fin (Module.finrank ℝ E)) :
63 g.metricInner p (chartMetricDual g α hp i) (chartMetricDual g α hp i) =
64 chartInvGramMatrix g α p i i := by
65 classical
66 rw [chartMetricDual_pairing]
67 simp [chartMetricDual, map_sum, map_smul, Module.Basis.repr_self, Finsupp.single_apply]
68
69theorem chartInvGramMatrix_diagonal_nonneg (g : RiemannianMetric I M) (α : M) {p : M}
70 (hp : p ∈ (trivializationAt E (TangentSpace I) α).baseSet)
71 (i : Fin (Module.finrank ℝ E)) : 0 ≤ chartInvGramMatrix g α p i i := by
72 rw [← chartMetricDual_sq g α hp i]
73 exact g.metricInner_self_nonneg p _

75/-- **Math.** Every chart coordinate is bounded by an explicit inverse-Gram factor times
76the actual metric norm, rather than the model-space norm. -/
77theorem abs_chartBasis_repr_le (g : RiemannianMetric I M) (α : M) {p : M}
78 (hp : p ∈ (trivializationAt E (TangentSpace I) α).baseSet)
79 (i : Fin (Module.finrank ℝ E)) (v : TangentSpace I p) :
80 |(chartBasisFamily α hp).repr v i| ≤
81 Real.sqrt (chartInvGramMatrix g α p i i) * AmbientBounds.metricNorm g p v := by
82 have h := metricInner_sq_le g p (chartMetricDual g α hp i) v
83 rw [chartMetricDual_pairing, chartMetricDual_sq] at h
84 have hs := Real.sqrt_le_sqrt h
85 simpa only [Real.sqrt_sq_eq_abs,
86 Real.sqrt_mul (chartInvGramMatrix_diagonal_nonneg g α hp i),
87 AmbientBounds.metricNorm] using hs
