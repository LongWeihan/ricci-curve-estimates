15/-- **Math.** Alternation kills every algebraic curvature form in dimension one. -/
16theorem algebraicCurvature_eq_zero_of_finrank_one
17 {V : Type*} [NormedAddCommGroup V] [InnerProductSpace ℝ V] [FiniteDimensional ℝ V]
18 {B : V → V → V → V → ℝ} (hB : Riemannian.IsAlgCurvatureForm B)
19 (hdim : Module.finrank ℝ V = 1) (x y z w : V) : B x y z w = 0 := by
20 by_cases hy : y = 0
21 · subst y
22 rw [hB.antisymm₁₂, hB.zero_left, neg_zero]
23 · obtain ⟨c, rfl⟩ := (finrank_eq_one_iff_of_nonzero' y hy).mp hdim x
24 rw [hB.smul_left, hB.self_left, mul_zero]
