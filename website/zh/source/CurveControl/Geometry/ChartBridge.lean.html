3/-!
4# Local coordinates of an actual curve flow
5
6These lemmas pass from the manifold-valued input to a fixed local chart near
7an interior spacetime point. The chart is local; no global chart or embedded
8image is assumed. The time coordinate is first.
9-/

25def chartCurveFamily (α : M) (c : ℝ → ℝ → M) (p : ℝ × ℝ) : E :=
26 extChartAt I α (c p.1 p.2)

35theorem contMDiff_curve {t : ℝ} (ht : t ∈ J) :
36 ContMDiff 𝓘(ℝ, ℝ) I ∞ (c t) := by
37 apply contMDiffOn_univ.mp
38 exact hc.smooth.comp (contMDiff_const.prodMk contMDiff_id).contMDiffOn
39 (by intro x _; exact ⟨ht, mem_univ x⟩)

41theorem contMDiffAt_family {p : ℝ × ℝ} (hp : p.1 ∈ interior J) :
42 ContMDiffAt (𝓘(ℝ, ℝ).prod 𝓘(ℝ, ℝ)) I ∞
43 (fun z : ℝ × ℝ => c z.1 z.2) p := by
44 exact hc.smooth.contMDiffAt
45 (prod_mem_nhds (mem_interior_iff_mem_nhds.mp hp) Filter.univ_mem)

47theorem contDiffAt_chartCurveFamily (α : M) {p : ℝ × ℝ}
48 (hp : p.1 ∈ interior J) (hα : c p.1 p.2 ∈ (chartAt H α).source) :
49 ContDiffAt ℝ ∞ (chartCurveFamily (I := I) α c) p := by
50 have h := (contMDiffAt_extChartAt' (I := I) hα).comp p (hc.contMDiffAt_family hp)
51 rw [← modelWithCornersSelf_prod, chartedSpaceSelf_prod] at h
52 exact h.contDiffAt

54theorem chartVelocity_eq_spatialFDeriv (α : M) {t x : ℝ}
55 (ht : t ∈ interior J) (hα : c t x ∈ (chartAt H α).source) :
56 chartFieldCoord (I := I) α (c t) (curveVelocity (I := I) (c t)) x =
57 fderiv ℝ (chartCurveFamily (I := I) α c) (t, x) (0, 1) := by
58 have hc' := hc.contMDiff_curve (interior_subset ht)
59 have hmem : ∀ᶠ y in 𝓝 x, c t y ∈ (chartAt H α).source :=
60 hc'.continuous.continuousAt ((chartAt H α).open_source.mem_nhds hα)
61 have hd := (hc.contDiffAt_chartCurveFamily α (p := (t, x)) ht hα).differentiableAt (by simp)
62 have hs : HasDerivAt (fun y : ℝ => (t, y)) (0, 1) x :=
63 (hasDerivAt_const x t).prodMk (hasDerivAt_id x)
64 exact chartFieldCoord_curveVelocity_eq hmem (hd.hasFDerivAt.comp_hasDerivAt x hs)

66theorem chartCurvature_eq_timeFDeriv (α : M) {t x : ℝ}
67 (ht : t ∈ interior J) (hα : c t x ∈ (chartAt H α).source) :
68 tangentCoordChange I (c t x) α (c t x) (curvatureVector (g t) (c t) x) =
69 fderiv ℝ (chartCurveFamily (I := I) α c) (t, x) (1, 0) := by
70 have hpath : ContinuousAt (fun s : ℝ => c s x) t :=
71 (hc.contMDiffAt_family ht).continuousAt.comp
72 (continuousAt_id.prodMk continuousAt_const)
73 have hmem : ∀ᶠ s in 𝓝 t, c s x ∈ (chartAt H α).source :=
74 hpath ((chartAt H α).open_source.mem_nhds hα)
75 have hd := (hc.contDiffAt_chartCurveFamily α (p := (t, x)) ht hα).differentiableAt (by simp)
76 have hs : HasDerivAt (fun s : ℝ => (s, x)) (1, 0) t :=
77 (hasDerivAt_id t).prodMk (hasDerivAt_const t x)
78 have hdtime : HasDerivAt (fun s => extChartAt I α (c s x))
79 (fderiv ℝ (chartCurveFamily (I := I) α c) (t, x) (1, 0)) t :=
80 by simpa only [Function.comp_def, chartCurveFamily] using
81 (hd.hasFDerivAt.comp_hasDerivAt t hs)
82 have hcoord := chartFieldCoord_curveVelocity_eq (I := I)
83 (x := α) (γ := fun s => c s x) hmem hdtime
84 change tangentCoordChange I (c t x) α (c t x)
85 (curveVelocity (I := I) (fun s => c s x) t) = _ at hcoord
86 rw [hc.timeVelocity_eq_curvatureVector ht x] at hcoord
87 exact hcoord
