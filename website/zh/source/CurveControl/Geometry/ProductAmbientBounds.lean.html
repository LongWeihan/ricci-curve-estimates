4/-!
5# Curvature and Ricci bounds independent of a one-dimensional factor
6
7The projection lengths are bounded by the actual product metric length. Actual
8intrinsic curvature/Ricci splitting and one-dimensional flatness then propagate
9the original constants. Covariant Ricci differentiation is a separate module.
10-/

26/-- **Math.** The two upstream pointwise curvature packages agree. -/
27theorem ambientRm_eq_canonical (g : Riemannian.RiemannianMetric I M) (p : M)
28 (x y z w : TangentSpace I p) : AmbientBounds.rm g p x y z w =
29 g.leviCivitaConnection.curvatureFormAt g p x y z w := by
30 unfold AmbientBounds.rm
31 rw [MorganTianLib.curvatureFormAt_def,
32 g.leviCivitaConnection.curvatureFormAt_eq g p
33 (X := extendVector p x) (Y := extendVector p y)
34 (Z := extendVector p z) (T := extendVector p w)]
35 all_goals simp

43/-- **Math.** First projection decreases actual Riemannian tangent length. -/
44theorem productL2_metricNorm_fst_le (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N)
45 (p : M × N) (x : TangentSpace (productL2Model I J) p) :
46 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm x).1 ≤
47 AmbientBounds.metricNorm (productL2Metric I J g h) p x := by
48 unfold AmbientBounds.metricNorm
49 rw [productL2Metric_metricInner_split]
50 exact Real.sqrt_le_sqrt (le_add_of_nonneg_right (h.metricInner_self_nonneg _ _))
51
52/-- **Math.** Second projection decreases actual Riemannian tangent length. -/
53theorem productL2_metricNorm_snd_le (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N)
54 (p : M × N) (x : TangentSpace (productL2Model I J) p) :

61/-- **Math.** The ambient curvature package used by curve estimates splits. -/
62theorem ambientRm_productL2 (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N)
63 (p : M × N) (x y z w : TangentSpace (productL2Model I J) p) :
64 AmbientBounds.rm (productL2Metric I J g h) p x y z w =
65 AmbientBounds.rm g p.1 (productL2Equiv.symm x).1 (productL2Equiv.symm y).1
66 (productL2Equiv.symm z).1 (productL2Equiv.symm w).1 +
67 AmbientBounds.rm h p.2 (productL2Equiv.symm x).2 (productL2Equiv.symm y).2
68 (productL2Equiv.symm z).2 (productL2Equiv.symm w).2 := by
69 simp only [ambientRm_eq_canonical]
70 exact canonicalCurvatureFormAt_productL2 g h p x y z w

72/-- **Math.** The Ricci bound propagates with precisely the same base constant
73for any one-dimensional factor metric. -/
74theorem productL2_ricci_bound_of_finrank_one
75 (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N) (p : M × N)
76 {B C D : ℝ} (hb : AmbientBounds.TensorBoundsAt g p.1 B C D)
77 (hdim : Module.finrank ℝ F = 1)
78 (x y : TangentSpace (productL2Model I J) p) :
79 |ricciTensorAt (productL2Metric I J g h) p x y| ≤
80 B * AmbientBounds.metricNorm (productL2Metric I J g h) p x *
81 AmbientBounds.metricNorm (productL2Metric I J g h) p y := by
82 rw [ricciTensorAt_productL2, ricciTensorAt_eq_zero_of_finrank_one hdim h, add_zero]
83 calc
84 _ ≤ B * AmbientBounds.metricNorm g p.1 (productL2Equiv.symm x).1 *
85 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm y).1 := hb.ricci _ _
86 _ ≤ _ := by
87 have hB := hb.ricci_nonneg
88 gcongr <;> first
89 | exact productL2_metricNorm_fst_le g h p _
90 | (unfold AmbientBounds.metricNorm; positivity)
91
92/-- **Math.** The full four-slot curvature bound propagates with precisely
93the same base constant for any one-dimensional factor metric. -/
94theorem productL2_curvature_bound_of_finrank_one
95 (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N) (p : M × N)
96 {B C D : ℝ} (hb : AmbientBounds.TensorBoundsAt g p.1 B C D)
97 (hdim : Module.finrank ℝ F = 1)
98 (x y z w : TangentSpace (productL2Model I J) p) :
99 |AmbientBounds.rm (productL2Metric I J g h) p x y z w| ≤
100 C * AmbientBounds.metricNorm (productL2Metric I J g h) p x *
101 AmbientBounds.metricNorm (productL2Metric I J g h) p y *
102 AmbientBounds.metricNorm (productL2Metric I J g h) p z *
103 AmbientBounds.metricNorm (productL2Metric I J g h) p w := by
104 rw [ambientRm_productL2, ambientRm_eq_canonical h,
105 canonicalCurvatureFormAt_eq_zero_of_finrank_one hdim h, add_zero]
106 calc
107 _ ≤ C * AmbientBounds.metricNorm g p.1 (productL2Equiv.symm x).1 *
108 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm y).1 *
109 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm z).1 *
110 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm w).1 := hb.curvature _ _ _ _
111 _ ≤ _ := by
112 have hC := hb.curvature_nonneg
113 gcongr <;> first
114 | exact productL2_metricNorm_fst_le g h p _
115 | (unfold AmbientBounds.metricNorm; positivity)
