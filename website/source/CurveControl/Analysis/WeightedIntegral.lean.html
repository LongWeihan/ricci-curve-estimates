17/-- Differentiation of a moving weight on a fixed parameter interval. The
18derivative bound is uniform on one time neighborhood and integrable in space. -/
19theorem hasDerivAt_weightedIntegral
20 {f v ft vt : ℝ → ℝ → ℝ} {t : ℝ} {s : Set ℝ} {bound : ℝ → ℝ}
21 (hs : s ∈ 𝓝 t)
22 (hmeas : ∀ᶠ u in 𝓝 t, AEStronglyMeasurable (fun x => f u x * v u x)
23 (volume.restrict (Ioc 0 1)))
24 (hint : Integrable (fun x => f t x * v t x) (volume.restrict (Ioc 0 1)))
25 (hdmeas : AEStronglyMeasurable (fun x => ft t x * v t x + f t x * vt t x)
26 (volume.restrict (Ioc 0 1)))
27 (hbound : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ s,
28 ‖ft u x * v u x + f u x * vt u x‖ ≤ bound x)
29 (hbi : Integrable bound (volume.restrict (Ioc 0 1)))
30 (hf : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ s,
31 HasDerivAt (fun z => f z x) (ft u x) u)
32 (hv : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ s,
33 HasDerivAt (fun z => v z x) (vt u x) u) :
34 HasDerivAt (fun u => ∫ x in (0 : ℝ)..1, f u x * v u x)
35 (∫ x in (0 : ℝ)..1, ft t x * v t x + f t x * vt t x) t := by
36 have hd : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ s,
37 HasDerivAt (fun z => f z x * v z x)
38 (ft u x * v u x + f u x * vt u x) u := by
39 filter_upwards [hf, hv] with x hfx hvx
40 intro u hu
41 exact (hfx u hu).mul (hvx u hu)
42 have h := (hasDerivAt_integral_of_dominated_loc_of_deriv_le hs
43 hmeas hint hdmeas hbound hbi hd).2
44 simpa only [intervalIntegral.integral_of_le (zero_le_one : (0 : ℝ) ≤ 1)] using h

46/-- Substitution of a proved density evolution law into the weighted first
47variation. This theorem does not assert or assume any unproved geometric law. -/
48theorem hasDerivAt_weightedIntegral_of_densityEvolution
49 {f v ft vt q r : ℝ → ℝ → ℝ} {t : ℝ}
50 (hfirst : HasDerivAt (fun u => ∫ x in (0 : ℝ)..1, f u x * v u x)
51 (∫ x in (0 : ℝ)..1, ft t x * v t x + f t x * vt t x) t)
52 (hevol : ∀ x ∈ Icc (0 : ℝ) 1, vt t x = -(q t x + r t x) * v t x) :
53 HasDerivAt (fun u => ∫ x in (0 : ℝ)..1, f u x * v u x)
54 (∫ x in (0 : ℝ)..1, (ft t x - f t x * (q t x + r t x)) * v t x) t := by
55 convert hfirst using 1
56 apply intervalIntegral.integral_congr
57 intro x hx
58 rw [uIcc_of_le (zero_le_one : (0 : ℝ) ≤ 1)] at hx
59 dsimp only
60 rw [hevol x hx]
61 simp only [sub_eq_add_neg, add_mul, mul_neg, neg_mul, mul_assoc]

63/-- Continuity of weighted integrals on a closed time interval follows from
64dominated convergence, requiring no time derivative at either endpoint. -/
65theorem continuousOn_weightedIntegral
66 {f v : ℝ → ℝ → ℝ} {a b : ℝ} {bound : ℝ → ℝ}
67 (hmeas : ∀ t ∈ Icc a b, AEStronglyMeasurable (fun x => f t x * v t x)
68 (volume.restrict (Ioc 0 1)))
69 (hbound : ∀ t ∈ Icc a b, ∀ᵐ x ∂volume.restrict (Ioc 0 1),
70 ‖f t x * v t x‖ ≤ bound x)
71 (hbi : Integrable bound (volume.restrict (Ioc 0 1)))
72 (hcont : ∀ᵐ x ∂volume.restrict (Ioc 0 1),
73 ContinuousOn (fun t => f t x * v t x) (Icc a b)) :
74 ContinuousOn (fun t => ∫ x in (0 : ℝ)..1, f t x * v t x) (Icc a b) := by
75 have h := continuousOn_of_dominated hmeas hbound hbi hcont
76 simpa only [intervalIntegral.integral_of_le (zero_le_one : (0 : ℝ) ≤ 1)] using h

78/-- Interior-time first variation. Only times strictly between `a` and `b`
79need derivatives; no smooth extension through the initial endpoint is needed.
80Together with `continuousOn_weightedIntegral` this supplies the analytic
81regularity used by closed-interval comparison arguments. -/
82theorem hasDerivAt_weightedIntegral_interior
83 {f v ft vt : ℝ → ℝ → ℝ} {a b t : ℝ} {bound : ℝ → ℝ}
84 (ht : t ∈ Ioo a b)
85 (hmeas : ∀ u ∈ Ioo a b, AEStronglyMeasurable (fun x => f u x * v u x)
86 (volume.restrict (Ioc 0 1)))
87 (hint : Integrable (fun x => f t x * v t x) (volume.restrict (Ioc 0 1)))
88 (hdmeas : AEStronglyMeasurable (fun x => ft t x * v t x + f t x * vt t x)
89 (volume.restrict (Ioc 0 1)))
90 (hbound : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ Ioo a b,
91 ‖ft u x * v u x + f u x * vt u x‖ ≤ bound x)
92 (hbi : Integrable bound (volume.restrict (Ioc 0 1)))
93 (hf : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ Ioo a b,
94 HasDerivAt (fun z => f z x) (ft u x) u)
95 (hv : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ Ioo a b,
96 HasDerivAt (fun z => v z x) (vt u x) u) :
97 HasDerivAt (fun u => ∫ x in (0 : ℝ)..1, f u x * v u x)
98 (∫ x in (0 : ℝ)..1, ft t x * v t x + f t x * vt t x) t := by
99 apply hasDerivAt_weightedIntegral (Ioo_mem_nhds ht.1 ht.2)
100 _ hint hdmeas hbound hbi hf hv
101 filter_upwards [Ioo_mem_nhds ht.1 ht.2] with u hu
102 exact hmeas u hu

104/-- Arclength differentiation in a fixed regular curve parameter. -/
105noncomputable def arcDeriv (v f : ℝ → ℝ) (x : ℝ) : ℝ := deriv f x / v x

107/-- A periodic flux derivative has zero integral, by the fundamental theorem
108of calculus. The zero integral is a conclusion, never a hypothesis. -/
109theorem integral_deriv_eq_zero_of_endpoints
110 {g : ℝ → ℝ}
111 (hdiff : ∀ x ∈ Icc (0 : ℝ) 1, DifferentiableAt ℝ g x)
112 (hint : IntervalIntegrable (deriv g) volume 0 1)
113 (hperiod : g 1 = g 0) :
114 (∫ x in (0 : ℝ)..1, deriv g x) = 0 := by
115 rw [intervalIntegral.integral_deriv_eq_sub
116 (by simpa only [uIcc_of_le (zero_le_one : (0 : ℝ) ≤ 1)] using hdiff) hint,
117 hperiod, sub_self]

119/-- Weighted spatial second derivatives integrate to zero when the actual
120first arclength derivative has matching endpoint values. The regularity and
121endpoint conditions must be verified by the periodic geometric caller. -/
122theorem integral_arcDeriv_arcDeriv_eq_zero
123 {f v : ℝ → ℝ}
124 (hv : ∀ x ∈ Icc (0 : ℝ) 1, 0 < v x)
125 (hdiff : ∀ x ∈ Icc (0 : ℝ) 1, DifferentiableAt ℝ (arcDeriv v f) x)
126 (hint : IntervalIntegrable (deriv (arcDeriv v f)) volume 0 1)
127 (hperiod : arcDeriv v f 1 = arcDeriv v f 0) :
128 (∫ x in (0 : ℝ)..1, arcDeriv v (arcDeriv v f) x * v x) = 0 := by
129 calc
130 _ = ∫ x in (0 : ℝ)..1, deriv (arcDeriv v f) x := by
131 apply intervalIntegral.integral_congr
132 intro x hx
133 rw [uIcc_of_le (zero_le_one : (0 : ℝ) ≤ 1)] at hx
134 exact div_mul_cancel₀ _ (ne_of_gt (hv x hx))
135 _ = 0 := integral_deriv_eq_zero_of_endpoints hdiff hint hperiod
