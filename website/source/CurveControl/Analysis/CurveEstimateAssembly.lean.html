6/-! # Complete scalar curve-estimate assembly
7
8This is an analysis theorem with explicit squared-curvature PDE and Kato
9inputs. Those two inputs and the density evolution must be proved by the
10geometry layer. Regularized PDEs, domination, integral differential
11inequalities, and regularization limits are all proved internally.
12Time derivatives are required only in the open time interval.
13-/

20structure ScalarCurveHypotheses (q v r P : ℝ → ℝ → ℝ) (a b A B : ℝ) : Prop where
21 q_smooth : ContDiffOn ℝ ∞ (Function.uncurry q) (Ioo a b ×ˢ univ)
22 v_smooth : ContDiffOn ℝ ∞ (Function.uncurry v) (Ioo a b ×ˢ univ)
23 q_continuous : ContinuousOn (Function.uncurry q) (Icc a b ×ˢ Icc (0 : ℝ) 1)
24 v_continuous : ContinuousOn (Function.uncurry v) (Icc a b ×ˢ Icc (0 : ℝ) 1)
25 q_nonneg : ∀ t ∈ Icc a b, ∀ x, 0 ≤ q t x
26 v_pos : ∀ t ∈ Icc a b, ∀ x, 0 < v t x
27 q_periodic : ∀ t ∈ Ioo a b, Function.Periodic (q t) 1
28 v_periodic : ∀ t ∈ Ioo a b, Function.Periodic (v t) 1
29 A_nonneg : 0 ≤ A
30 B_nonneg : 0 ≤ B
31 ricci_bound : ∀ t ∈ Ioo a b, ∀ x, |r t x| ≤ B
32 density_evolution : ∀ t ∈ Ioo a b, ∀ x,
33 deriv (fun u => v u x) t = -(q t x + r t x) * v t x
34 kato : ∀ t ∈ Ioo a b, ∀ x,
35 (arcDeriv (v t) (q t) x) ^ 2 ≤ 4 * q t x * (P t x) ^ 2
36 squared_pde : ∀ t ∈ Ioo a b, ∀ x,
37 deriv (fun u => q u x) t ≤ arcDeriv (v t) (arcDeriv (v t) (q t)) x -
38 2 * (P t x) ^ 2 + 2 * (q t x) ^ 2 + A * (q t x + Real.sqrt (q t x))

201/-- The complete scalar estimate, with a constant independent of the
202regularization parameter. The initial time needs only rectangle continuity.
203Here `weightedCurvature (q t) (v t)` is the actual integral `∫₀¹ √q v`. -/
204theorem ScalarCurveHypotheses.curve_estimate
205 {q v r P : ℝ → ℝ → ℝ} {a b A B : ℝ}
206 (H : ScalarCurveHypotheses q v r P a b A B) :
207 ∀ t ∈ Icc a b,
208 (∫ x in (0 : ℝ)..1, v t x) ≤
209 (∫ x in (0 : ℝ)..1, v a x) * Real.exp (B * (t - a)) ∧
210 weightedCurvature (q t) (v t) + (∫ x in (0 : ℝ)..1, v t x) ≤
211 (weightedCurvature (q a) (v a) + (∫ x in (0 : ℝ)..1, v a x)) *
212 Real.exp ((A / 2 + B) * (t - a)) := by
213 intro t ht
214 have ha : a ∈ Icc a b := ⟨le_rfl, ht.1.trans ht.2⟩
215 constructor
216 · exact (H.regularized_comparison (ε := 1) zero_lt_one t ht).1
217 · apply weightedCurvature_exp_bound
218 (continuousOn_space_slice H.q_continuous ht)
219 (continuousOn_space_slice H.v_continuous ht)
220 (fun x _ => H.q_nonneg t ht x) (fun x _ => (H.v_pos t ht x).le)
221 (continuousOn_space_slice H.q_continuous ha)
222 (continuousOn_space_slice H.v_continuous ha)
223 (fun x _ => H.q_nonneg a ha x) (fun x _ => (H.v_pos a ha x).le)
224 exact fun ε hε => (H.regularized_comparison hε t ht).2
