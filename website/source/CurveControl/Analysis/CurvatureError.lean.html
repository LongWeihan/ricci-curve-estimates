5/-!
6# Absorbing the corrected curvature error
7
8This is the scalar contraction estimate for the spatial-connection version of
9the Morgan--Tian correction. The geometric caller must supply bounds on the
10actual Ricci, curvature and covariant-Ricci contractions. These assumptions are
11not a substitute for proving their occurrence in the geometric evolution law.
12-/
13
14namespace CurveControl.Analysis
15
16/-- A single coefficient controlling the quadratic and linear curvature errors. -/
17def curvatureErrorCoefficient (B K D : ℝ) : ℝ := max (6 * B + 2 * K) (6 * D)
18
19theorem curvatureErrorCoefficient_nonneg {B K D : ℝ} (hD : 0 ≤ D) :
20 0 ≤ curvatureErrorCoefficient B K D :=
21 le_trans (by linarith) (le_max_right _ _)
22
23/-- The correction terms are bounded by a quadratic and a linear term. -/
24theorem corrected_curvature_error_le
25 {k B K D rSS rHH rHSHS dSSH dHSS : ℝ}
26 (hk : 0 ≤ k)
27 (hSS : |rSS| ≤ B) (hHH : |rHH| ≤ B * k ^ 2)
28 (hRm : |rHSHS| ≤ K * k ^ 2)
29 (hDS : |dSSH| ≤ D * k) (hDH : |dHSS| ≤ D * k) :
30 4 * k ^ 2 * rSS - 2 * rHH + 2 * rHSHS - 4 * dSSH + 2 * dHSS ≤
31 (6 * B + 2 * K) * k ^ 2 + 6 * D * k := by
32 have hSS' := (abs_le.mp hSS).2
33 have hHH' := (abs_le.mp hHH).1
34 have hRm' := (abs_le.mp hRm).2
35 have hDS' := (abs_le.mp hDS).1
36 have hDH' := (abs_le.mp hDH).2
37 have hprod := mul_le_mul_of_nonneg_left hSS' (sq_nonneg k)
38 nlinarith
39
40theorem corrected_curvature_error_abs_le
41 {k B K D rSS rHH rHSHS dSSH dHSS : ℝ}
42 (hk : 0 ≤ k)
43 (hSS : |rSS| ≤ B) (hHH : |rHH| ≤ B * k ^ 2)
44 (hRm : |rHSHS| ≤ K * k ^ 2)
45 (hDS : |dSSH| ≤ D * k) (hDH : |dHSS| ≤ D * k) :
46 |4 * k ^ 2 * rSS - 2 * rHH + 2 * rHSHS - 4 * dSSH + 2 * dHSS| ≤
47 (6 * B + 2 * K) * k ^ 2 + 6 * D * k := by
48 apply abs_le.mpr
49 constructor
50 · have h := corrected_curvature_error_le (rSS := -rSS) (rHH := -rHH)
51 (rHSHS := -rHSHS) (dSSH := -dSSH) (dHSS := -dHSS) hk
52 (by simpa using hSS) (by simpa using hHH) (by simpa using hRm)
53 (by simpa using hDS) (by simpa using hDH)
54 linarith
55 · exact corrected_curvature_error_le hk hSS hHH hRm hDS hDH

57theorem corrected_curvature_error_le_coefficient
58 {k B K D rSS rHH rHSHS dSSH dHSS : ℝ}
59 (hk : 0 ≤ k)
60 (hSS : |rSS| ≤ B) (hHH : |rHH| ≤ B * k ^ 2)
61 (hRm : |rHSHS| ≤ K * k ^ 2)
62 (hDS : |dSSH| ≤ D * k) (hDH : |dHSS| ≤ D * k) :
63 4 * k ^ 2 * rSS - 2 * rHH + 2 * rHSHS - 4 * dSSH + 2 * dHSS ≤
64 curvatureErrorCoefficient B K D * (k ^ 2 + k) := by
65 apply (corrected_curvature_error_le hk hSS hHH hRm hDS hDH).trans
66 have hq := mul_le_mul_of_nonneg_right (le_max_left (6 * B + 2 * K) (6 * D))
67 (sq_nonneg k)
68 have hl := mul_le_mul_of_nonneg_right (le_max_right (6 * B + 2 * K) (6 * D)) hk
69 dsimp [curvatureErrorCoefficient]
70 nlinarith
