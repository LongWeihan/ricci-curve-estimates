6# Removing scalar regularization after integral comparison

25theorem regularizedWeightedCurvature_sub_bounds {q v : ℝ → ℝ} {ε : ℝ}

27 (hq0 : ∀ s ∈ Icc 0 1, 0 ≤ q s) (hv0 : ∀ s ∈ Icc 0 1, 0 ≤ v s)

30 regularizedWeightedCurvature q v ε - weightedCurvature q v ≤

33 (hq.add continuousOn_const).sqrt

35 (hc.mul hv).intervalIntegrable_of_Icc (by norm_num : (0 : ℝ) ≤ 1)
36 have hj : IntervalIntegrable (fun s => Real.sqrt (q s) * v s) volume 0 1 :=
37 (hq.sqrt.mul hv).intervalIntegrable_of_Icc (by norm_num : (0 : ℝ) ≤ 1)
38 have hk : IntervalIntegrable (fun s => ε * v s) volume 0 1 :=
39 (continuousOn_const.mul hv).intervalIntegrable_of_Icc
40 (by norm_num : (0 : ℝ) ≤ 1)
41 have heq : regularizedWeightedCurvature q v ε - weightedCurvature q v =
42 ∫ s in (0 : ℝ)..1, (regularizedNorm (q s) ε * v s - Real.sqrt (q s) * v s) :=
43 (intervalIntegral.integral_sub hi hj).symm
