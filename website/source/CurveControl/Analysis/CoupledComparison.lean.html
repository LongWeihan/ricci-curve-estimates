3/-!
4# Abstract coupled exponential comparison
5
6This file adapts mathlib's Grönwall comparison engine to the two scalar quantities
7used for length and regularized total curvature. It is an analysis component only:
8the geometric application must prove continuity, the indicated right derivatives,
9and both differential inequalities from the geometric evolution equations.
10In particular none of these hypotheses is a proved curve-flow evolution formula here.
11
12The comparison works for arbitrary real coefficients and signed scalar functions.
13The geometric assumptions `0 ≤ L` and `0 ≤ c₁` are therefore not needed at this
14stage (they can be needed when deriving the differential inequalities).
15-/
16

21/-- Scalar exponential comparison, with a right derivative on the half-open time
22interval. This is a specialization of mathlib's Grönwall engine, with zero forcing. -/
23theorem le_exp_of_deriv_right_le
24 {f f' : ℝ → ℝ} {a b K C : ℝ}
25 (hf : ContinuousOn f (Icc a b))
26 (hf' : ∀ t ∈ Ico a b, HasDerivWithinAt f (f' t) (Ici t) t)
27 (hinit : f a ≤ C)
28 (hbound : ∀ t ∈ Ico a b, f' t ≤ K * f t) :
29 ∀ t ∈ Icc a b, f t ≤ C * exp (K * (t - a)) := by
30 intro t ht
31 have h := le_gronwallBound_of_liminf_deriv_right_le (ε := 0) hf
32 (fun x hx _ hr => (hf' x hx).liminf_right_slope_le hr) hinit
33 (fun x hx => by simpa only [add_zero] using hbound x hx) t ht
34 simpa only [gronwallBound_ε0] using h

36/-- Coupled length/curvature comparison with their actual initial values.
37All geometric differential inequalities remain explicit hypotheses; this theorem
38does not assert that arbitrary curve flows satisfy them. -/
39theorem coupled_exp_comparison_right
40 {L Q L' Q' : ℝ → ℝ} {a b c₁ c₂ : ℝ}
41 (hL : ContinuousOn L (Icc a b)) (hQ : ContinuousOn Q (Icc a b))
42 (hL' : ∀ t ∈ Ico a b, HasDerivWithinAt L (L' t) (Ici t) t)
43 (hQ' : ∀ t ∈ Ico a b, HasDerivWithinAt Q (Q' t) (Ici t) t)
44 (hLbound : ∀ t ∈ Ico a b, L' t ≤ c₂ * L t)
45 (hQbound : ∀ t ∈ Ico a b, Q' t ≤ (c₁ + c₂) * Q t + c₁ * L t) :
46 ∀ t ∈ Icc a b,
47 L t ≤ L a * exp (c₂ * (t - a)) ∧
48 Q t + L t ≤ (Q a + L a) * exp ((c₁ + c₂) * (t - a)) := by
49 have hsum : ∀ t ∈ Ico a b,
50 Q' t + L' t ≤ (c₁ + c₂) * (Q t + L t) := by
51 intro t ht
52 calc
53 Q' t + L' t ≤ ((c₁ + c₂) * Q t + c₁ * L t) + c₂ * L t :=
54 add_le_add (hQbound t ht) (hLbound t ht)
55 _ = (c₁ + c₂) * (Q t + L t) := by ring
56 exact fun t ht =>
57 ⟨le_exp_of_deriv_right_le hL hL' le_rfl hLbound t ht,
58 le_exp_of_deriv_right_le (hQ.add hL)
59 (fun x hx => (hQ' x hx).add (hL' x hx)) le_rfl hsum t ht⟩

61/-- Classical-derivative interface to the coupled comparison. -/
62theorem coupled_exp_comparison
63 {L Q L' Q' : ℝ → ℝ} {a b c₁ c₂ : ℝ}
64 (hL : ContinuousOn L (Icc a b)) (hQ : ContinuousOn Q (Icc a b))
65 (hL' : ∀ t ∈ Ico a b, HasDerivAt L (L' t) t)
66 (hQ' : ∀ t ∈ Ico a b, HasDerivAt Q (Q' t) t)
67 (hLbound : ∀ t ∈ Ico a b, L' t ≤ c₂ * L t)
68 (hQbound : ∀ t ∈ Ico a b, Q' t ≤ (c₁ + c₂) * Q t + c₁ * L t) :
69 ∀ t ∈ Icc a b,
70 L t ≤ L a * exp (c₂ * (t - a)) ∧
71 Q t + L t ≤ (Q a + L a) * exp ((c₁ + c₂) * (t - a)) :=
72 coupled_exp_comparison_right hL hQ
73 (fun t ht => (hL' t ht).hasDerivWithinAt)
74 (fun t ht => (hQ' t ht).hasDerivWithinAt) hLbound hQbound

76/-- Arbitrary indexed families have the same exponential bounds whenever the
77coefficients, initial time, and initial upper bounds are independent of the index.
78No topology, finiteness, or compactness is imposed on the index type. -/
79theorem family_coupled_exp_comparison_right
80 {ι : Type*} {L Q L' Q' : ι → ℝ → ℝ} {a b c₁ c₂ Cₗ Cₛ : ℝ}
81 (hL : ∀ i, ContinuousOn (L i) (Icc a b))
82 (hQ : ∀ i, ContinuousOn (Q i) (Icc a b))
83 (hL' : ∀ i t, t ∈ Ico a b → HasDerivWithinAt (L i) (L' i t) (Ici t) t)
84 (hQ' : ∀ i t, t ∈ Ico a b → HasDerivWithinAt (Q i) (Q' i t) (Ici t) t)
85 (hLbound : ∀ i t, t ∈ Ico a b → L' i t ≤ c₂ * L i t)
86 (hQbound : ∀ i t, t ∈ Ico a b →
87 Q' i t ≤ (c₁ + c₂) * Q i t + c₁ * L i t)
88 (hLinit : ∀ i, L i a ≤ Cₗ)
89 (hSumInit : ∀ i, Q i a + L i a ≤ Cₛ) :
90 ∀ i t, t ∈ Icc a b →
91 L i t ≤ Cₗ * exp (c₂ * (t - a)) ∧
92 Q i t + L i t ≤ Cₛ * exp ((c₁ + c₂) * (t - a)) := by
93 intro i t ht
94 obtain ⟨hl, hs⟩ := coupled_exp_comparison_right (hL i) (hQ i)
95 (hL' i) (hQ' i) (hLbound i) (hQbound i) t ht
96 exact ⟨hl.trans (mul_le_mul_of_nonneg_right (hLinit i) (exp_pos _).le),
97 hs.trans (mul_le_mul_of_nonneg_right (hSumInit i) (exp_pos _).le)⟩
98
