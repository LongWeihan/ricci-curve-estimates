4/-!
5# Abstract energy integral comparison
6
7These are scalar analysis lemmas, not curve-flow evolution theorems. The geometric
8caller must prove the length differential inequality and identify its energy.
9Only continuity of length on the closed interval and actual right derivatives in
10its interior are required. There is no assumption of endpoint differentiability
11or integrability of the length derivative. Energy need only be integrable.
12-/

18/-- Integrating-factor energy estimate. The right derivative is needed only in
19the open interval. This weighted estimate does not require sign assumptions. -/
20theorem weighted_energy_integral_le
21 {L L' E : ℝ → ℝ} {s t B : ℝ}
22 (hst : s ≤ t) (hL : ContinuousOn L (Icc s t))
23 (hL' : ∀ r ∈ Ioo s t, HasDerivWithinAt L (L' r) (Ioi r) r)
24 (hE : IntegrableOn E (Icc s t))
25 (hbound : ∀ r ∈ Ioo s t, L' r ≤ B * L r - E r) :
26 (∫ r in s..t, exp (-B * (r - s)) * E r) ≤
27 L s - exp (-B * (t - s)) * L t := by
28 let w : ℝ → ℝ := fun r => exp (-B * (r - s))
29 have hw : ∀ r, HasDerivAt w (-B * w r) r := by
30 intro r
31 convert (((hasDerivAt_id r).sub_const s).const_mul (-B)).exp using 1 <;>
32 simp [w, mul_comm]
33 have hwcont : ContinuousOn w (Icc s t) :=
34 (continuous_iff_continuousAt.mpr fun r => (hw r).continuousAt).continuousOn
35 have hWE : IntegrableOn (fun r => w r * E r) (Icc s t) :=
36 hE.continuousOn_mul hwcont isCompact_Icc
37 have hd : ∀ r ∈ Ioo s t,
38 HasDerivWithinAt (fun x => -(w x * L x))
39 (w r * (B * L r - L' r)) (Ioi r) r := by
40 intro r hr
41 have hd0 : HasDerivWithinAt (fun x => -(w x * L x))
42 (-((-B * w r) * L r + w r * L' r)) (Ioi r) r :=
43 ((hw r).hasDerivWithinAt.mul (hL' r hr)).neg
44 convert hd0 using 1
45 ring
46 have h := intervalIntegral.integral_le_sub_of_hasDeriv_right_of_le hst
47 (hwcont.mul hL).neg hd hWE (fun r hr =>
48 mul_le_mul_of_nonneg_left (by linarith [hbound r hr]) (exp_pos _).le)
49 dsimp [w] at h
50 simp only [sub_self, mul_zero, exp_zero, one_mul] at h
51 linarith

53/-- For nonnegative energy and nonnegative growth coefficient, the weighted
54estimate gives the unweighted accumulated-energy bound. Length itself need not
55be assumed nonnegative in this abstract implication. -/
56theorem energy_integral_le
57 {L L' E : ℝ → ℝ} {s t B : ℝ}
58 (hst : s ≤ t) (hL : ContinuousOn L (Icc s t))
59 (hL' : ∀ r ∈ Ioo s t, HasDerivWithinAt L (L' r) (Ioi r) r)
60 (hE : IntegrableOn E (Icc s t))
61 (hbound : ∀ r ∈ Ioo s t, L' r ≤ B * L r - E r)
62 (hB : 0 ≤ B) (hEpos : ∀ r ∈ Icc s t, 0 ≤ E r) :
63 (∫ r in s..t, E r) ≤ exp (B * (t - s)) * L s - L t := by
64 have hwcont : ContinuousOn (fun r => exp (-B * (r - s))) (Icc s t) := by
65 fun_prop
66 have hWE := hE.continuousOn_mul hwcont isCompact_Icc
67 have hEi : IntervalIntegrable E volume s t := by
68 exact (intervalIntegrable_iff_integrableOn_Icc_of_le hst).mpr hE
69 have hWEi : IntervalIntegrable (fun r => exp (-B * (r - s)) * E r) volume s t := by
70 exact (intervalIntegrable_iff_integrableOn_Icc_of_le hst).mpr hWE
71 have hcompare : (∫ r in s..t, E r) ≤
72 exp (B * (t - s)) * (∫ r in s..t, exp (-B * (r - s)) * E r) := by
73 rw [← intervalIntegral.integral_const_mul]
74 apply intervalIntegral.integral_mono_on hst hEi (hWEi.const_mul _)
75 intro r hr
76 have he : 1 ≤ exp (B * (t - s)) * exp (-B * (r - s)) := by
77 rw [← exp_add, one_le_exp_iff]
78 nlinarith [mul_nonneg hB (sub_nonneg.mpr hr.2)]
79 calc
80 E r = 1 * E r := (one_mul _).symm
81 _ ≤ (exp (B * (t - s)) * exp (-B * (r - s))) * E r :=
82 mul_le_mul_of_nonneg_right he (hEpos r hr)
83 _ = exp (B * (t - s)) * (exp (-B * (r - s)) * E r) := mul_assoc _ _ _
84 have hweighted := weighted_energy_integral_le hst hL hL' hE hbound
85 have hexp : exp (B * (t - s)) * exp (-B * (t - s)) = 1 := by
86 rw [← exp_add]
87 have hz : B * (t - s) + -B * (t - s) = 0 := by ring
88 rw [hz, exp_zero]
89 calc
90 (∫ r in s..t, E r) ≤
91 exp (B * (t - s)) * (∫ r in s..t, exp (-B * (r - s)) * E r) := hcompare
92 _ ≤ exp (B * (t - s)) * (L s - exp (-B * (t - s)) * L t) :=
93 mul_le_mul_of_nonneg_left hweighted (exp_pos _).le
94 _ = exp (B * (t - s)) * L s - L t := by rw [mul_sub, ← mul_assoc, hexp, one_mul]
