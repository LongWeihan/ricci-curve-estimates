48/-- Cancellation of the negative gradient term after regularization. The

50theorem regularized_kato_cancellation {q qs P ε : ℝ}
51 (hq : 0 ≤ q) (hε : 0 < ε) (hK : qs ^ 2 ≤ 4 * q * P ^ 2) :
52 qs ^ 2 / (4 * regularizedNorm q ε ^ 3) ≤ P ^ 2 / regularizedNorm q ε := by
53 have hp := regularizedNorm_pos hq hε
54 have hn := ne_of_gt hp

64/-- Pure scalar consequence of the explicitly supplied evolution and Kato
65inequalities. Here `qt`, `qs`, `qss` are scalar slots, not hidden derivatives. -/
66theorem regularized_pde_algebra {q qt qs qss P A ε : ℝ}
67 (hq : 0 ≤ q) (hε : 0 < ε) (hA : 0 ≤ A)
68 (hK : qs ^ 2 ≤ 4 * q * P ^ 2)
69 (hPDE : qt ≤ qss - 2 * P ^ 2 + 2 * q ^ 2 + A * (q + Real.sqrt q)) :
70 qt / (2 * regularizedNorm q ε) ≤
71 qss / (2 * regularizedNorm q ε) - qs ^ 2 / (4 * regularizedNorm q ε ^ 3) +
72 (Real.sqrt q) ^ 3 + (A / 2) * (regularizedNorm q ε + 1) := by
73 have hp := regularizedNorm_pos hq hε
74 have hn := ne_of_gt hp
75 have hd := div_le_div_of_nonneg_right hPDE
76 (le_of_lt (mul_pos (by norm_num : (0 : ℝ) < 2) hp))
77 have heq : (qss - 2 * P ^ 2 + 2 * q ^ 2 + A * (q + Real.sqrt q)) /
78 (2 * regularizedNorm q ε) =
79 qss / (2 * regularizedNorm q ε) - P ^ 2 / regularizedNorm q ε +
80 q ^ 2 / regularizedNorm q ε +
81 (A / 2) * (q / regularizedNorm q ε + Real.sqrt q / regularizedNorm q ε) := by
82 field_simp
83 rw [heq] at hd

85 (add_le_add (div_regularizedNorm_le hq hε) (sqrt_div_regularizedNorm_le_one hq hε))
