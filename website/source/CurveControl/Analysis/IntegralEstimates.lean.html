18/-- Explicit analytic hypotheses for the fixed-parameter differentiation
19theorem. This packages local derivative and domination data, not a conclusion
20about the derivative of the whole integral. -/
21structure MovingWeightData (f v ft vt : ℝ → ℝ → ℝ) (t : ℝ) where
22 timeSet : Set ℝ
23 bound : ℝ → ℝ
24 neighborhood : timeSet ∈ 𝓝 t
25 measurable : ∀ᶠ u in 𝓝 t, AEStronglyMeasurable (fun x => f u x * v u x)
26 (volume.restrict (Ioc 0 1))
27 integrable : Integrable (fun x => f t x * v t x) (volume.restrict (Ioc 0 1))
28 derivative_measurable : AEStronglyMeasurable
29 (fun x => ft t x * v t x + f t x * vt t x) (volume.restrict (Ioc 0 1))
30 derivative_bound : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ timeSet,
31 ‖ft u x * v u x + f u x * vt u x‖ ≤ bound x
32 bound_integrable : Integrable bound (volume.restrict (Ioc 0 1))
33 function_derivative : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ timeSet,
34 HasDerivAt (fun z => f z x) (ft u x) u
35 weight_derivative : ∀ᵐ x ∂volume.restrict (Ioc 0 1), ∀ u ∈ timeSet,
36 HasDerivAt (fun z => v z x) (vt u x) u

46/-- The favorable cubic reaction is absorbed by the negative density term. -/
47theorem sqrt_cube_sub_regularized_mul_nonpos {q ε : ℝ} (hq : 0 ≤ q) :
48 Real.sqrt q ^ 3 - regularizedNorm q ε * q ≤ 0 := by
49 have hm := mul_le_mul_of_nonneg_right (sqrt_le_regularizedNorm q ε) hq
50 have hs := Real.sq_sqrt hq
51 calc
52 _ = Real.sqrt q * q - regularizedNorm q ε * q := by rw [pow_succ, hs]; ring
53 _ ≤ 0 := sub_nonpos.mpr hm

55theorem regularized_density_pointwise_le {q ε htime diffusion r B C : ℝ}
56 (hq : 0 ≤ q) (hr : |r| ≤ B)
57 (hpde : htime ≤ diffusion + Real.sqrt q ^ 3 + C * (regularizedNorm q ε + 1)) :
58 htime - regularizedNorm q ε * (q + r) ≤
59 diffusion + (C + B) * regularizedNorm q ε + C := by

65/-- The length variation retains its full negative squared-curvature energy. -/
66theorem integral_density_le {q r v : ℝ → ℝ} {B : ℝ}
67 (hv : ∀ x ∈ Icc (0 : ℝ) 1, 0 ≤ v x)
68 (hr : ∀ x ∈ Icc (0 : ℝ) 1, |r x| ≤ B)
69 (ileft : IntervalIntegrable (fun x => -(q x + r x) * v x) volume 0 1)
70 (iv : IntervalIntegrable v volume 0 1)
71 (iqv : IntervalIntegrable (fun x => q x * v x) volume 0 1) :
72 (∫ x in (0 : ℝ)..1, -(q x + r x) * v x) ≤
73 B * (∫ x in (0 : ℝ)..1, v x) - ∫ x in (0 : ℝ)..1, q x * v x := by
74 have hm : (∫ x in (0 : ℝ)..1, -(q x + r x) * v x) ≤
75 ∫ x in (0 : ℝ)..1, B * v x - q x * v x := by
76 apply intervalIntegral.integral_mono_on zero_le_one ileft ((iv.const_mul B).sub iqv)
77 intro x hx
78 have hb := mul_le_mul_of_nonneg_right (abs_le.mp (hr x hx)).1 (hv x hx)
79 nlinarith
80 simpa only [intervalIntegral.integral_sub (iv.const_mul B) iqv,
81 intervalIntegral.integral_const_mul] using hm

83/-- An integrated regularized PDE, with an explicit diffusion integral.
84The following periodic version proves that integral vanishes by FTC. -/
85theorem integral_regularized_density_le {q r v htime diffusion : ℝ → ℝ} {ε B C : ℝ}
86 (hq : ∀ x ∈ Icc (0 : ℝ) 1, 0 ≤ q x)
87 (hv : ∀ x ∈ Icc (0 : ℝ) 1, 0 ≤ v x)
88 (hr : ∀ x ∈ Icc (0 : ℝ) 1, |r x| ≤ B)
89 (hpde : ∀ x ∈ Icc (0 : ℝ) 1,
90 htime x ≤ diffusion x + Real.sqrt (q x) ^ 3 + C * (regularizedNorm (q x) ε + 1))
91 (ileft : IntervalIntegrable
92 (fun x => (htime x - regularizedNorm (q x) ε * (q x + r x)) * v x) volume 0 1)
93 (idiff : IntervalIntegrable (fun x => diffusion x * v x) volume 0 1)
94 (ihv : IntervalIntegrable (fun x => regularizedNorm (q x) ε * v x) volume 0 1)
95 (iv : IntervalIntegrable v volume 0 1) :
96 (∫ x in (0 : ℝ)..1, (htime x - regularizedNorm (q x) ε * (q x + r x)) * v x) ≤
97 (∫ x in (0 : ℝ)..1, diffusion x * v x) +
98 (C + B) * (∫ x in (0 : ℝ)..1, regularizedNorm (q x) ε * v x) +
99 C * (∫ x in (0 : ℝ)..1, v x) := by
100 have hm : (∫ x in (0 : ℝ)..1,
101 (htime x - regularizedNorm (q x) ε * (q x + r x)) * v x) ≤
102 ∫ x in (0 : ℝ)..1,
103 (diffusion x * v x + (C + B) * (regularizedNorm (q x) ε * v x)) + C * v x := by
104 apply intervalIntegral.integral_mono_on zero_le_one ileft
105 ((idiff.add (ihv.const_mul (C + B))).add (iv.const_mul C))
106 intro x hx
107 have h := mul_le_mul_of_nonneg_right
108 (regularized_density_pointwise_le (hq x hx) (hr x hx) (hpde x hx)) (hv x hx)
109 calc
110 _ ≤ (diffusion x + (C + B) * regularizedNorm (q x) ε + C) * v x := h
111 _ = _ := by ring
112 simpa only [intervalIntegral.integral_add (idiff.add (ihv.const_mul (C + B)))
113 (iv.const_mul C), intervalIntegral.integral_add idiff (ihv.const_mul (C + B)),
114 intervalIntegral.integral_const_mul] using hm
