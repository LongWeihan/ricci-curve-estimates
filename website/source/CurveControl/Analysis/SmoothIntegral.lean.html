39/-- An intermediate rectangle criterion: continuous derivative fields and
40actual slice derivatives supply every domination/measurability requirement. -/
41noncomputable def movingWeightData_of_rectangle
42 {f v ft vt : ℝ → ℝ → ℝ} {a b t : ℝ}
43 (ht : Icc a b ∈ 𝓝 t)
44 (hf : ContinuousOn (Function.uncurry f) (Icc a b ×ˢ Icc (0 : ℝ) 1))
45 (hv : ContinuousOn (Function.uncurry v) (Icc a b ×ˢ Icc (0 : ℝ) 1))
46 (hft : ContinuousOn (Function.uncurry ft) (Icc a b ×ˢ Icc (0 : ℝ) 1))
47 (hvt : ContinuousOn (Function.uncurry vt) (Icc a b ×ˢ Icc (0 : ℝ) 1))
48 (hfd : ∀ u ∈ Icc a b, ∀ x ∈ Icc (0 : ℝ) 1, HasDerivAt (fun z => f z x) (ft u x) u)
49 (hvd : ∀ u ∈ Icc a b, ∀ x ∈ Icc (0 : ℝ) 1, HasDerivAt (fun z => v z x) (vt u x) u) :
50 MovingWeightData f v ft vt t := by
51 apply Classical.choice
52 have hdc := (hft.mul hv).add (hf.mul hvt)
53 have hti : t ∈ Icc a b := mem_of_mem_nhds ht
54 obtain ⟨M, hM⟩ := (isCompact_Icc.prod isCompact_Icc).bddAbove_image hdc.norm
55 refine ⟨?_⟩
56 refine ⟨Icc a b, (fun _ => M), ht, ?_, ?_, ?_, ?_, ?_, ?_, ?_⟩
57 · filter_upwards [ht] with u hu
58 exact ((continuousOn_space_slice hf hu).mul (continuousOn_space_slice hv hu)).mono
59 Ioc_subset_Icc_self |>.aestronglyMeasurable measurableSet_Ioc
60 · exact (ContinuousOn.intervalIntegrable_of_Icc zero_le_one
61 ((continuousOn_space_slice hf hti).mul (continuousOn_space_slice hv hti))).1
62 · exact (((continuousOn_space_slice hft hti).mul (continuousOn_space_slice hv hti)).add
63 ((continuousOn_space_slice hf hti).mul (continuousOn_space_slice hvt hti))).mono
64 Ioc_subset_Icc_self |>.aestronglyMeasurable measurableSet_Ioc

74theorem hasDerivAt_time_slice_of_contDiffOn {f : ℝ → ℝ → ℝ} {U : Set ℝ}
75 (hU : IsOpen U) (hf : ContDiffOn ℝ 1 (Function.uncurry f) (U ×ˢ univ))
76 {u x : ℝ} (hu : u ∈ U) :
77 HasDerivAt (fun z => f z x) (fderiv ℝ (Function.uncurry f) (u, x) (1, 0)) u := by
78 have hd := (hf.differentiableOn (by simp) (u, x) ⟨hu, mem_univ x⟩).differentiableAt ((hU.prod isOpen_univ).mem_nhds ⟨hu, mem_univ x⟩)
79 have he : HasDerivAt (fun z : ℝ => (z, x)) ((1, 0) : ℝ × ℝ) u :=
80 (hasDerivAt_id u).prodMk (hasDerivAt_const u x)
81 exact hd.hasFDerivAt.comp_hasDerivAt u he

91/-- Joint C1 regularity on an open time strip automatically supplies the
92moving-weight data, using a compact time neighborhood and compact space. -/
93noncomputable def movingWeightData_of_contDiffOn {f v : ℝ → ℝ → ℝ} {U : Set ℝ} {t : ℝ}
94 (hU : IsOpen U) (ht : t ∈ U)
95 (hf : ContDiffOn ℝ 1 (Function.uncurry f) (U ×ˢ univ))
96 (hv : ContDiffOn ℝ 1 (Function.uncurry v) (U ×ˢ univ)) :
97 MovingWeightData f v (fun u x => deriv (fun z => f z x) u)
98 (fun u x => deriv (fun z => v z x) u) t := by
99 apply Classical.choice
100 obtain ⟨a, b, _, hn, hsub⟩ := exists_Icc_mem_subset_of_mem_nhds (hU.mem_nhds ht)
101 have hs : Icc a b ×ˢ Icc (0 : ℝ) 1 ⊆ U ×ˢ univ :=
102 fun _ hp => ⟨hsub hp.1, mem_univ _⟩
103 refine ⟨?_⟩
104 apply movingWeightData_of_rectangle
105 (ft := fun u x => deriv (fun z => f z x) u)
106 (vt := fun u x => deriv (fun z => v z x) u)
107 hn (hf.continuousOn.mono hs) (hv.continuousOn.mono hs)
108 ((continuousOn_time_deriv_of_contDiffOn hU hf).mono hs)
109 ((continuousOn_time_deriv_of_contDiffOn hU hv).mono hs)
110 · intro u hu x _
111 exact (hasDerivAt_time_slice_of_contDiffOn hU hf (hsub hu)).differentiableAt.hasDerivAt
112 · intro u hu x _
113 exact (hasDerivAt_time_slice_of_contDiffOn hU hv (hsub hu)).differentiableAt.hasDerivAt

117theorem hasDerivAt_weightedIntegral_of_contDiffOn {f v : ℝ → ℝ → ℝ} {U : Set ℝ} {t : ℝ}
118 (hU : IsOpen U) (ht : t ∈ U)
