3/-!
4# Applying the curve estimates to an actual scaled static factor
5
6Caller contract: supply an actual smooth one-dimensional manifold and an actual
7Riemannian metric h₀ on it. This file constructs hλ = λ² h₀ for every positive
8real λ, including its positivity, bounded-unit-ball and smoothness proofs.
9If the supplied manifold is a smooth circle, this is its usual metric scaling.
10No circle atlas, closed-curve immersion or curve-shortening-flow existence is
11asserted by this example. Those are explicit geometric caller inputs.
12
13The final theorem bounds length and total curvature simultaneously for every
14positive scale, curve index and time in the full closed time interval. Its
15constants are chosen from the compact base flow before the scale or curves.
16-/

28/-- **Math.** A positive real scale multiplies an actual metric by its square.
29Every metric-structure field is constructed, not supplied as an assumption. -/
30def scaledFactorMetric (h₀ : Riemannian.RiemannianMetric J N)
31 (ℓ : {r : ℝ // 0 < r}) : Riemannian.RiemannianMetric J N where
32 inner p := (ℓ.val ^ 2) • h₀.inner p
33 symm p u v := by
34 change ℓ.val ^ 2 * h₀.inner p u v = ℓ.val ^ 2 * h₀.inner p v u
35 rw [h₀.symm]
36 pos p u hu := by
37 change 0 < ℓ.val ^ 2 * h₀.inner p u u
38 exact mul_pos (sq_pos_of_pos ℓ.property) (h₀.pos p u hu)
39 isVonNBounded p := by
40 apply isVonNBounded_of_posDef (E := F)
41 intro u hu
42 change 0 < ℓ.val ^ 2 * h₀.inner p u u
43 exact mul_pos (sq_pos_of_pos ℓ.property) (h₀.pos p u hu)
44 contMDiff := h₀.contMDiff.const_smul_section

85/-- **Math.** Concrete scaled-metric application. Exponential constants are
86chosen from the compact base flow before h₀, the positive scale, curve indices
87or initial bounds. Smooth immersed closed product-flow existence and the two
88common initial bounds are the caller's only curve inputs. -/
89theorem exists_scaledFactor_curve_bounds [CompactSpace M]
90 {g : ℝ → Riemannian.RiemannianMetric I M} {a b : ℝ}
91 (hg : IsRicciFlowOn g (Icc a b)) (hab : a ≤ b) (hdim : Module.finrank ℝ F = 1) :
92 ∃ B C : ℝ, 0 ≤ B ∧ 0 ≤ C ∧
93 ∀ (h₀ : Riemannian.RiemannianMetric J N) (ι : Type*)
94 (c : {r : ℝ // 0 < r} → ι → ℝ → ℝ → M × N),
95 (∀ ℓ i, IsCurveShorteningFlowOn (I := productL2Model I J)
96 (fun t => productL2Metric I J (g t) (scaledFactorMetric h₀ ℓ)) (c ℓ i) (Icc a b)) →
97 ∀ L₀ Θ₀ : ℝ,
98 (∀ ℓ i, curveLength (productL2Metric I J (g a) (scaledFactorMetric h₀ ℓ))
99 (c ℓ i a) ≤ L₀) →
100 (∀ ℓ i, totalCurvature (productL2Metric I J (g a) (scaledFactorMetric h₀ ℓ))
101 (c ℓ i a) ≤ Θ₀) →
102 ∀ ℓ i, ∀ t ∈ Icc a b,
103 curveLength (productL2Metric I J (g t) (scaledFactorMetric h₀ ℓ)) (c ℓ i t) ≤
104 L₀ * Real.exp (B * (t - a)) ∧
105 totalCurvature (productL2Metric I J (g t) (scaledFactorMetric h₀ ℓ)) (c ℓ i t) ≤
106 (Θ₀ + L₀) * Real.exp (C * (t - a)) := by
107 obtain ⟨B, C, hB, hC, hestimate⟩ :=
108 exists_uniform_positive_scale_product_estimate I J (N := N) hg hab hdim
109 refine ⟨B, C, hB, hC, ?_⟩
110 intro h₀ ι c hc L₀ Θ₀ hL₀ hΘ₀
111 exact hestimate ι (scaledFactorMetric h₀) c hc L₀ Θ₀ hL₀ hΘ₀

113/-- **Math.** A downstream full-interval bound suitable for a later ramp
114argument: one finite length bound and one finite total-curvature bound hold
115simultaneously for all t, all curves and every positive scale, even as λ→0.
116This is an estimate only; it does not assert ramp or curve-flow existence. -/
117theorem exists_scaledFactor_fullInterval_bounds [CompactSpace M]
118 {g : ℝ → Riemannian.RiemannianMetric I M} {a b : ℝ}
119 (hg : IsRicciFlowOn g (Icc a b)) (hab : a ≤ b) (hdim : Module.finrank ℝ F = 1) :
120 ∃ B C : ℝ, 0 ≤ B ∧ 0 ≤ C ∧
121 ∀ (h₀ : Riemannian.RiemannianMetric J N) (ι : Type*)
122 (c : {r : ℝ // 0 < r} → ι → ℝ → ℝ → M × N),
123 (∀ ℓ i, IsCurveShorteningFlowOn (I := productL2Model I J)
124 (fun t => productL2Metric I J (g t) (scaledFactorMetric h₀ ℓ)) (c ℓ i) (Icc a b)) →
125 ∀ L₀ Θ₀ : ℝ,
126 (∀ ℓ i, curveLength (productL2Metric I J (g a) (scaledFactorMetric h₀ ℓ))
127 (c ℓ i a) ≤ L₀) →
128 (∀ ℓ i, totalCurvature (productL2Metric I J (g a) (scaledFactorMetric h₀ ℓ))
129 (c ℓ i a) ≤ Θ₀) →
130 ∀ ℓ i, ∀ t ∈ Icc a b,
131 curveLength (productL2Metric I J (g t) (scaledFactorMetric h₀ ℓ)) (c ℓ i t) ≤
132 L₀ * Real.exp (B * (b - a)) ∧
133 totalCurvature (productL2Metric I J (g t) (scaledFactorMetric h₀ ℓ)) (c ℓ i t) ≤
134 (Θ₀ + L₀) * Real.exp (C * (b - a)) := by
135 obtain ⟨B, C, hB, hC, hestimate⟩ :=
136 exists_scaledFactor_curve_bounds (I := I) (J := J) (N := N) hg hab hdim
137 refine ⟨B, C, hB, hC, ?_⟩
138 intro h₀ ι c hc L₀ Θ₀ hL₀ hΘ₀ ℓ i t ht
139 have h := hestimate h₀ ι c hc L₀ Θ₀ hL₀ hΘ₀ ℓ i t ht
140 have hLa : 0 ≤ curveLength (productL2Metric I J (g a) (scaledFactorMetric h₀ ℓ))
141 (c ℓ i a) := by
142 apply intervalIntegral.integral_nonneg zero_le_one
143 intro x _
144 exact curveSpeed_nonneg _ _ _
145 have hΘa : 0 ≤ totalCurvature (productL2Metric I J (g a) (scaledFactorMetric h₀ ℓ))
146 (c ℓ i a) := by
147 apply intervalIntegral.integral_nonneg zero_le_one
148 intro x _
149 exact mul_nonneg (curvature_nonneg _ _ _) (curveSpeed_nonneg _ _ _)
150 have hLpos : 0 ≤ L₀ := hLa.trans (hL₀ ℓ i)
151 have hΘpos : 0 ≤ Θ₀ := hΘa.trans (hΘ₀ ℓ i)
152 have htime := sub_le_sub_right ht.2 a
153 exact ⟨h.1.trans (mul_le_mul_of_nonneg_left
154 (Real.exp_le_exp.mpr (mul_le_mul_of_nonneg_left htime hB)) hLpos),
155 h.2.trans (mul_le_mul_of_nonneg_left
156 (Real.exp_le_exp.mpr (mul_le_mul_of_nonneg_left htime hC)) (add_nonneg hΘpos hLpos))⟩
