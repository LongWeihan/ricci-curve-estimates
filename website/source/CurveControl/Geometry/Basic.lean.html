5/-!
6Actual metric quantities for a parametrized curve. All derivatives are the

59/-- Curvature vector `∇ₛS = speed⁻¹ ∇ₓS`. -/
60def curvatureVector (g : RiemannianMetric I M) (γ : ℝ → M)
61 (x : ℝ) : TangentSpace I (γ x) :=
62 (curveSpeed g γ x)⁻¹ • covDerivAlong g γ (unitTangent g γ) x

64/-- Squared norm of the curvature vector. -/
65def curvatureSq (g : RiemannianMetric I M) (γ : ℝ → M) (x : ℝ) : ℝ :=
66 g.metricInner (γ x) (curvatureVector g γ x) (curvatureVector g γ x)

75/-- Length over one parameter period; integrability follows later from smoothness. -/
76def curveLength (g : RiemannianMetric I M) (γ : ℝ → M) : ℝ :=
