3/-!
4# Genuine smooth product metric and projection estimates
5
6We reuse DoCarmo's `DCProductMetric`, constructed as the sum of the pullbacks
7along the two smooth projections. Its positive definiteness and smoothness in
8the actual tangent-bundle trivializations are proved upstream, not assumed here.
9The tangent space is that of the product model `I.prod J`, not an auxiliary
10bundle with a stipulated splitting. No curvature or connection splitting is
11claimed. The construction permits any finite-dimensional second factor and
12any of its Riemannian metrics, including scaled circle metrics.
13-/

29/-- **Math.** Length of an actual tangent vector in a specified Riemannian metric. -/
30def metricVectorLength (g : RiemannianMetric I M) (p : M)
31 (v : TangentSpace I p) : ℝ := Real.sqrt (g.metricInner p v v)
32
33theorem metricVectorLength_nonneg (g : RiemannianMetric I M) (p : M)
34 (v : TangentSpace I p) : 0 ≤ metricVectorLength g p v := Real.sqrt_nonneg _
35
36theorem metricVectorLength_sq (g : RiemannianMetric I M) (p : M)
37 (v : TangentSpace I p) : (metricVectorLength g p v) ^ 2 = g.metricInner p v v :=
38 Real.sq_sqrt (g.metricInner_self_nonneg p v)

42/-- **Math.** The smooth positive-definite metric on the real product manifold. -/
43def productMetric (g : RiemannianMetric I M) (h : RiemannianMetric J N) :
44 RiemannianMetric (I.prod J) (M × N) := DCProductMetric g h

46/-- **Math.** Metric splitting evaluated on the genuine product tangent space. -/
47theorem productMetric_inner (g : RiemannianMetric I M) (h : RiemannianMetric J N)
48 (p : M × N) (u v : TangentSpace (I.prod J) p) :
49 (productMetric g h).metricInner p u v =
50 g.metricInner p.1 u.1 v.1 + h.metricInner p.2 u.2 v.2 := by
51 change DCProductForm g h p u v = _
52 simp only [DCProductForm, add_apply, DCInducedForm_apply,
53 mfderiv_fst, mfderiv_snd]
54 rfl

56/-- **Math.** The square of product length is the sum of the squared factor lengths. -/
57theorem productMetric_length_sq (g : RiemannianMetric I M) (h : RiemannianMetric J N)
58 (p : M × N) (u : TangentSpace (I.prod J) p) :
59 (metricVectorLength (productMetric g h) p u) ^ 2 =
60 (metricVectorLength g p.1 u.1) ^ 2 + (metricVectorLength h p.2 u.2) ^ 2 := by
61 simp only [metricVectorLength_sq, productMetric_inner]

63/-- **Math.** First projection is length decreasing, uniformly in the second-factor metric. -/
64theorem productMetric_fst_length_le (g : RiemannianMetric I M) (h : RiemannianMetric J N)
65 (p : M × N) (u : TangentSpace (I.prod J) p) :
66 metricVectorLength g p.1 u.1 ≤ metricVectorLength (productMetric g h) p u := by
67 unfold metricVectorLength
68 rw [productMetric_inner]
69 apply Real.sqrt_le_sqrt
70 exact le_add_of_nonneg_right (h.metricInner_self_nonneg p.2 u.2)
