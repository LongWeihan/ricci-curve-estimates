28theorem periodic_deriv {F : Type*} [NormedAddCommGroup F] [NormedSpace ℝ F]
29 {f : ℝ → F} (hf : Function.Periodic f 1) : Function.Periodic (deriv f) 1 := by
30 intro x

32 have heq : (fun y => f (y + 1)) = f := funext hf
33 rw [heq]

142 have hdu : ContDiffAt ℝ ∞
143 (fun z => fderiv ℝ (chartCurveFamily (I := I) α c) z d) p :=
144 (hu.fderiv_right (by simp)).clm_apply contDiffAt_const
145 have hG (i j : Fin (Module.finrank ℝ E)) :
146 ContDiffAt ℝ ∞ (fun z : ℝ × ℝ =>
147 chartGramOnE (I := I) (g z.1) α i j (chartCurveFamily (I := I) α c z)) p := by
148 have hy : chartCurveFamily (I := I) α c p ∈ (extChartAt I α).target :=

150 have houter := (contDiffOn_chartGramOnE_timeSpace hg α i j).contDiffAt
151 (prod_mem_nhds (mem_interior_iff_mem_nhds.mp hp)
152 ((isOpen_extChartAt_target (I := I) α).mem_nhds hy))
153 exact houter.comp p (contDiffAt_fst.prodMk hu)

350 z.1 ∈ J ∧ c z.1 z.2 ∈ (chartAt H (c p.1 p.2)).source := by
351 have htime : ∀ᶠ z : ℝ × ℝ in 𝓝[J ×ˢ (univ : Set ℝ)] p, z.1 ∈ J :=
352 Filter.Eventually.mono self_mem_nhdsWithin (fun _ hz => hz.1)
353 have hspace : ∀ᶠ z : ℝ × ℝ in 𝓝[J ×ˢ (univ : Set ℝ)] p,
354 c z.1 z.2 ∈ (chartAt H (c p.1 p.2)).source :=
355 (hc.smooth p ⟨hp, mem_univ _⟩).continuousWithinAt
356 ((chartAt H (c p.1 p.2)).open_source.mem_nhds (mem_chart_source H _))
357 exact htime.and hspace
