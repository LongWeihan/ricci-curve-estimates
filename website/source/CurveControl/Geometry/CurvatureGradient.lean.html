36/-- **Math.** Cauchy--Schwarz for the actual metric, obtained from its positivity. -/
37theorem metricInner_sq_le (g : RiemannianMetric I M) (p : M)
38 (V W : TangentSpace I p) :
39 (g.metricInner p V W) ^ 2 ≤ g.metricInner p V V * g.metricInner p W W := by
40 by_cases hV : V = 0
41 · simp [hV]
42 have hpos := g.metricInner_self_pos p V hV
43 have hn := g.metricInner_self_nonneg p
44 (W - (g.metricInner p V W / g.metricInner p V V) • V)
45 simp only [g.metricInner_sub_left, g.metricInner_sub_right,
46 g.metricInner_smul_left, g.metricInner_smul_right,
47 g.metricInner_comm p W V] at hn
48 have heq : g.metricInner p V V *
49 (g.metricInner p W W - (g.metricInner p V W / g.metricInner p V V) * g.metricInner p V W -
50 (g.metricInner p V W / g.metricInner p V V) *
51 (g.metricInner p V W - (g.metricInner p V W / g.metricInner p V V) * g.metricInner p V V)) =
52 g.metricInner p V V * g.metricInner p W W - (g.metricInner p V W) ^ 2 := by
53 field_simp
54 ring
55 have hm := mul_nonneg (le_of_lt hpos) hn
56 have hh : 0 ≤ g.metricInner p V V * g.metricInner p W W - (g.metricInner p V W) ^ 2 := by
57 rw [← heq]
58 nlinarith only [hm]
59 linarith
