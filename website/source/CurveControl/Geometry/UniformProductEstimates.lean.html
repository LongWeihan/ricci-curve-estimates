38/-- **Math.** Norm comparison for the covariant-Ricci product identity.
39The identity argument is supplied by the independent genuine product-tensor
40splitting theorem in the closed wrapper below. -/
41theorem productL2_covRic_bound_of_split
42 (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N)
43 (p : M × N) {B R D : ℝ} (hb : AmbientBounds.TensorBoundsAt g p.1 B R D)
44 (hdim : Module.finrank ℝ F = 1)
45 (hSplit : ∀ u v w : TangentSpace (productL2Model I J) p,
46 AmbientBounds.covRic (productL2Metric I J g h) p u v w =
47 AmbientBounds.covRic g p.1 (productL2Equiv.symm u).1
48 (productL2Equiv.symm v).1 (productL2Equiv.symm w).1 +
49 AmbientBounds.covRic h p.2 (productL2Equiv.symm u).2
50 (productL2Equiv.symm v).2 (productL2Equiv.symm w).2)
51 (u v w : TangentSpace (productL2Model I J) p) :
52 |AmbientBounds.covRic (productL2Metric I J g h) p u v w| ≤
53 D * AmbientBounds.metricNorm (productL2Metric I J g h) p u *
54 AmbientBounds.metricNorm (productL2Metric I J g h) p v *
55 AmbientBounds.metricNorm (productL2Metric I J g h) p w := by
56 rw [hSplit, ambientCovRic_eq_zero_of_finrank_one J hdim h, add_zero]
57 calc
58 _ ≤ D * AmbientBounds.metricNorm g p.1 (productL2Equiv.symm u).1 *
59 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm v).1 *
60 AmbientBounds.metricNorm g p.1 (productL2Equiv.symm w).1 := hb.covRic _ _ _
61 _ ≤ _ := by
62 have hD := hb.covRic_nonneg
63 gcongr <;> first
64 | exact productL2_metricNorm_fst_le g h p _
65 | (unfold AmbientBounds.metricNorm; positivity)

85/-- **Math.** All ambient tensor bounds of a genuine product with any static
86one-dimensional metric are inherited from the base with unchanged constants.

88theorem productL2_tensorBounds_of_finrank_one
89 (g : Riemannian.RiemannianMetric I M) (h : Riemannian.RiemannianMetric J N)
90 (p : M × N) {B R D : ℝ} (hb : AmbientBounds.TensorBoundsAt g p.1 B R D)

105/-- **Math.** Product-family estimates from uniform actual product tensors.
106Ricci-flow status of every product is proved from the base flow and the static
107one-dimensional factor, rather than supplied as a hypothesis. -/
108theorem static_product_family_estimate_of_tensorBounds {Λ ι : Type*}
109 {g : ℝ → Riemannian.RiemannianMetric I M} {a b B R D L₀ Θ₀ : ℝ}
110 (hg : IsRicciFlowOn g (Icc a b)) (hab : a ≤ b) (hdim : Module.finrank ℝ F = 1)
111 (h : Λ → Riemannian.RiemannianMetric J N) (c : Λ → ι → ℝ → ℝ → M × N)
112 (hc : ∀ ℓ i, IsCurveShorteningFlowOn (I := productL2Model I J)
113 (fun t => productL2Metric I J (g t) (h ℓ)) (c ℓ i) (Icc a b))
114 (hBounds : ∀ ℓ, AmbientBounds.UniformTensorBoundsOn
115 (fun t => productL2Metric I J (g t) (h ℓ)) (Icc a b) B R D)
116 (hL₀ : ∀ ℓ i, curveLength (productL2Metric I J (g a) (h ℓ)) (c ℓ i a) ≤ L₀)
117 (hΘ₀ : ∀ ℓ i, totalCurvature (productL2Metric I J (g a) (h ℓ)) (c ℓ i a) ≤ Θ₀) :
118 ∀ ℓ i, ∀ t ∈ Icc a b,

121 totalCurvature (productL2Metric I J (g t) (h ℓ)) (c ℓ i t) ≤
122 (Θ₀ + L₀) * Real.exp ((max (6 * B + 2 * R) (6 * D) / 2 + B) * (t - a)) := by

125 have hresult := curve_family_uniform_estimate (hc ℓ) hp hab (Subset.refl _)
126 (hBounds ℓ) (hL₀ ℓ) (fun i => add_le_add (hΘ₀ ℓ i) (hL₀ ℓ i)) i t ht

130 exact curveSpeed_nonneg _ _ _

143/-- **Math.** Compact-base uniform product estimate. The constants are chosen
144before the arbitrary external metric index, all static factor metrics, all
145curve indices and the initial bounds. Thus neither shrinking/scaling the
146factor metric nor selecting another curve changes the exponential constants.
147The one-dimensional factor is genuine supplied manifold geometry, not an
148assumed circle atlas. -/
149theorem exists_uniform_static_product_family_estimate [CompactSpace M]
150 {g : ℝ → Riemannian.RiemannianMetric I M} {a b : ℝ}
151 (hg : IsRicciFlowOn g (Icc a b)) (hab : a ≤ b) (hdim : Module.finrank ℝ F = 1) :
152 ∃ B C : ℝ, 0 ≤ B ∧ 0 ≤ C ∧
153 ∀ (Λ ι : Type*) (h : Λ → Riemannian.RiemannianMetric J N)
154 (c : Λ → ι → ℝ → ℝ → M × N),
155 (∀ ℓ i, IsCurveShorteningFlowOn (I := productL2Model I J)
156 (fun t => productL2Metric I J (g t) (h ℓ)) (c ℓ i) (Icc a b)) →
157 ∀ L₀ Θ₀ : ℝ,
158 (∀ ℓ i, curveLength (productL2Metric I J (g a) (h ℓ)) (c ℓ i a) ≤ L₀) →
159 (∀ ℓ i, totalCurvature (productL2Metric I J (g a) (h ℓ)) (c ℓ i a) ≤ Θ₀) →
160 ∀ ℓ i, ∀ t ∈ Icc a b,
161 curveLength (productL2Metric I J (g t) (h ℓ)) (c ℓ i t) ≤
162 L₀ * Real.exp (B * (t - a)) ∧
163 totalCurvature (productL2Metric I J (g t) (h ℓ)) (c ℓ i t) ≤
164 (Θ₀ + L₀) * Real.exp (C * (t - a)) := by
165 obtain ⟨B, R, D, hB, hR, hD, hBounds⟩ := compact_base_uniform_coefficients I hg
166 have hA : 0 ≤ max (6 * B + 2 * R) (6 * D) :=
167 le_max_of_le_left (by linarith)
168 refine ⟨B, max (6 * B + 2 * R) (6 * D) / 2 + B, hB, by positivity, ?_⟩
169 intro Λ ι h c hc L₀ Θ₀ hL₀ hΘ₀
170 exact static_product_family_estimate_of_tensorBounds I J hg hab hdim h c hc
171 (fun ℓ => uniformTensorBoundsOn_productL2_oneDimensional I J hBounds (h ℓ) hdim)
172 hL₀ hΘ₀

174/-- **Math.** Positive-scale-indexed corollary. The supplied metrics may, for
175example, be scaled metrics on an independently supplied smooth circle. No
176lower bound on the scale, nor any scale-regularity hypothesis, is required.
177All exponential coefficients are chosen before the entire metric family. -/
178theorem exists_uniform_positive_scale_product_estimate [CompactSpace M]
179 {g : ℝ → Riemannian.RiemannianMetric I M} {a b : ℝ}
180 (hg : IsRicciFlowOn g (Icc a b)) (hab : a ≤ b) (hdim : Module.finrank ℝ F = 1) :
181 ∃ B C : ℝ, 0 ≤ B ∧ 0 ≤ C ∧

183 (c : {ℓ : ℝ // 0 < ℓ} → ι → ℝ → ℝ → M × N),
184 (∀ ℓ i, IsCurveShorteningFlowOn (I := productL2Model I J)
185 (fun t => productL2Metric I J (g t) (h ℓ)) (c ℓ i) (Icc a b)) →
186 ∀ L₀ Θ₀ : ℝ,
187 (∀ ℓ i, curveLength (productL2Metric I J (g a) (h ℓ)) (c ℓ i a) ≤ L₀) →
188 (∀ ℓ i, totalCurvature (productL2Metric I J (g a) (h ℓ)) (c ℓ i a) ≤ Θ₀) →
189 ∀ ℓ i, ∀ t ∈ Icc a b,
190 curveLength (productL2Metric I J (g t) (h ℓ)) (c ℓ i t) ≤
