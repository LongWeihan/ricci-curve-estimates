14theorem trilinear_basis_expand (L : V →ₗ[ℝ] V →ₗ[ℝ] V →ₗ[ℝ] ℝ)
15 (b : Module.Basis ι ℝ V) (u v w : V) :
16 L u v w = ∑ i, ∑ j, ∑ k,
17 (b.repr u i) * (b.repr v j) * (b.repr w k) * L (b i) (b j) (b k) := by
18 classical
19 calc
20 L u v w = L (∑ i, (b.repr u i) • b i)
21 (∑ j, (b.repr v j) • b j) (∑ k, (b.repr w k) • b k) := by
22 rw [b.sum_repr u, b.sum_repr v, b.sum_repr w]
23 _ = _ := by
24 simp only [map_sum, map_smul, LinearMap.sum_apply, LinearMap.smul_apply,
25 smul_eq_mul, Finset.mul_sum, mul_assoc]
26 rw [Finset.sum_comm]
27 conv_lhs =>
28 arg 2
29 ext j
30 rw [Finset.sum_comm]
31 rw [Finset.sum_comm]
32 apply Finset.sum_congr rfl
33 intro i _
34 apply Finset.sum_congr rfl
35 intro j _
36 apply Finset.sum_congr rfl
37 intro k _
38 ring

40theorem quadrilinear_basis_expand (L : V →ₗ[ℝ] V →ₗ[ℝ] V →ₗ[ℝ] V →ₗ[ℝ] ℝ)
41 (b : Module.Basis ι ℝ V) (u v w z : V) :
42 L u v w z = ∑ i, ∑ j, ∑ k, ∑ l,
43 (b.repr u i) * (b.repr v j) * (b.repr w k) * (b.repr z l) *
44 L (b i) (b j) (b k) (b l) := by
45 classical
46 calc
47 L u v w z = L (∑ i, (b.repr u i) • b i)
48 (∑ j, (b.repr v j) • b j) (∑ k, (b.repr w k) • b k)
49 (∑ l, (b.repr z l) • b l) := by
50 rw [b.sum_repr u, b.sum_repr v, b.sum_repr w, b.sum_repr z]
51 _ = _ := by
52 simp only [map_sum, map_smul, LinearMap.sum_apply, LinearMap.smul_apply,
53 smul_eq_mul, Finset.mul_sum, mul_assoc]
54 rw [Finset.sum_comm]
55 conv_lhs =>
56 arg 2
57 ext k
58 rw [Finset.sum_comm]
59 arg 2
60 ext j
61 rw [Finset.sum_comm]
62 rw [Finset.sum_comm]
63 conv_lhs =>
64 arg 2
65 ext j
66 rw [Finset.sum_comm]
67 rw [Finset.sum_comm]
68 apply Finset.sum_congr rfl
69 intro i _
70 apply Finset.sum_congr rfl
71 intro j _
72 apply Finset.sum_congr rfl
73 intro k _
74 apply Finset.sum_congr rfl
75 intro l _

85/-- Currying of the actual covariant derivative of Ricci in its three slots. -/
86def covRicciTrilinear (g : Riemannian.RiemannianMetric I M)
87 (nabla : AffineConnection I M) (hLC : nabla.IsLeviCivita g) (p : M) :
88 TangentSpace I p →ₗ[ℝ] TangentSpace I p →ₗ[ℝ] TangentSpace I p →ₗ[ℝ] ℝ where
89 toFun := covRicciTensorBilin g nabla hLC p
90 map_add' := by
91 intro u v
92 ext w z
93 exact covRicciAt_add_dir g nabla hLC p u v w z
94 map_smul' := by
95 intro a u
96 ext v w
97 exact covRicciAt_smul_dir g nabla hLC p a u v w

104/-- Works for every finite basis, in particular `chartBasisFamily α hp`. -/
105theorem covRicciAt_basis_expand (g : Riemannian.RiemannianMetric I M)
106 (nabla : AffineConnection I M) (hLC : nabla.IsLeviCivita g) (p : M)
107 {ι : Type*} [Fintype ι] (b : Module.Basis ι ℝ (TangentSpace I p))
108 (u v w : TangentSpace I p) :
109 covRicciAt g nabla hLC p u v w = ∑ i, ∑ j, ∑ k,
110 (b.repr u i) * (b.repr v j) * (b.repr w k) *
111 covRicciAt g nabla hLC p (b i) (b j) (b k) :=
112 trilinear_basis_expand (covRicciTrilinear g nabla hLC p) b u v w

114/-- Currying of the intrinsic Morgan–Tian curvature form in its four slots. -/
115def curvatureQuadrilinear (g : Riemannian.RiemannianMetric I M)
116 (nabla : AffineConnection I M) (p : M) :
117 TangentSpace I p →ₗ[ℝ] TangentSpace I p →ₗ[ℝ]
118 TangentSpace I p →ₗ[ℝ] TangentSpace I p →ₗ[ℝ] ℝ where
119 toFun := fun u => {
120 toFun := fun v => LinearMap.mk₂ ℝ
121 (fun w z => MorganTianLib.curvatureFormAt g nabla p u v w z)
122 (fun w w' z => curvatureFormAt_add_trd g nabla p u v w w' z)
123 (fun a w z => curvatureFormAt_smul_trd g nabla p a u v w z)
124 (fun w z z' => curvatureFormAt_add_fth g nabla p u v w z z')
125 (fun a w z => curvatureFormAt_smul_fth g nabla p a u v w z)
126 map_add' := by
127 intro v v'
128 ext w z
129 exact curvatureFormAt_add_snd g nabla p u v v' w z
130 map_smul' := by
131 intro a v
132 ext w z
133 exact curvatureFormAt_smul_snd g nabla p a u v w z }
134 map_add' := by
135 intro u u'
136 ext v w z
137 exact curvatureFormAt_add_left g nabla p u u' v w z
138 map_smul' := by
139 intro a u
140 ext v w z
141 exact curvatureFormAt_smul_left g nabla p a u v w z

148 (nabla : AffineConnection I M) (p : M)
