23/-- The Ricci trace is invariant under a genuine linear isometry. -/
24theorem ricciForm_isometry
25 {B : V → V → V → V → ℝ} {C : E → E → E → E → ℝ}
26 (hB : IsAlgCurvatureForm B) (hC : IsAlgCurvatureForm C)
27 (e : V ≃ₗᵢ[ℝ] E)
28 (h : ∀ x y z w, C (e x) (e y) (e z) (e w) = B x y z w)
29 (x y : V) : ricciForm hC (e x) (e y) = ricciForm hB x y := by
30 classical
31 let b := stdOrthonormalBasis ℝ V
32 rw [ricciForm_eq_sum hC _ _ (b.map e), ricciForm_eq_sum hB _ _ b]
33 apply Finset.sum_congr rfl
34 intro i _
35 simpa only [OrthonormalBasis.map_apply] using h x (b i) y (b i)
36
37/-- Tracing a curvature form that splits over an orthogonal product gives the
38sum of the two factor Ricci forms. The isometry prevents a model norm from being
39silently substituted for the Riemannian metric. -/
40theorem ricciForm_product
41 {B : V → V → V → V → ℝ}
42 {B₁ : E → E → E → E → ℝ} {B₂ : F → F → F → F → ℝ}
43 (hB : IsAlgCurvatureForm B) (h₁ : IsAlgCurvatureForm B₁)
44 (h₂ : IsAlgCurvatureForm B₂)
45 (e : V ≃ₗᵢ[ℝ] WithLp 2 (E × F))
46 (h : ∀ x y z w,
47 B x y z w =
48 B₁ (e x).ofLp.1 (e y).ofLp.1 (e z).ofLp.1 (e w).ofLp.1 +
49 B₂ (e x).ofLp.2 (e y).ofLp.2 (e z).ofLp.2 (e w).ofLp.2)
50 (x y : V) :
51 ricciForm hB x y =
52 ricciForm h₁ (e x).ofLp.1 (e y).ofLp.1 +
53 ricciForm h₂ (e x).ofLp.2 (e y).ofLp.2 := by
54 classical
55 let b₁ := stdOrthonormalBasis ℝ E
56 let b₂ := stdOrthonormalBasis ℝ F
57 rw [ricciForm_eq_sum hB _ _ ((b₁.prod b₂).map e.symm),
58 ricciForm_eq_sum h₁ _ _ b₁, ricciForm_eq_sum h₂ _ _ b₂,
59 Fintype.sum_sum_type]
60 congr 1 <;> apply Finset.sum_congr rfl <;> intro i _
61 · simp only [OrthonormalBasis.map_apply, h, LinearIsometryEquiv.apply_symm_apply,
62 OrthonormalBasis.prod_apply, Sum.elim_inl, Function.comp_apply,
63 LinearMap.inl_apply, WithLp.ofLp_toLp, h₂.zero_two, add_zero]
64 · simp only [OrthonormalBasis.map_apply, h, LinearIsometryEquiv.apply_symm_apply,
65 OrthonormalBasis.prod_apply, Sum.elim_inr, Function.comp_apply,
66 LinearMap.inr_apply, WithLp.ofLp_toLp, h₁.zero_two, zero_add]
