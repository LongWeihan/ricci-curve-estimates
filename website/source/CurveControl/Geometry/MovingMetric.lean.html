13/-- **Math.** Restrict a jointly differentiable scalar function to the diagonal. -/
14theorem hasDerivAt_diagonal_of_slices {F : ℝ × ℝ → ℝ} {t a b : ℝ}
15 (hF : DifferentiableAt ℝ F (t, t))
16 (ha : HasDerivAt (fun s => F (s, t)) a t)
17 (hb : HasDerivAt (fun s => F (t, s)) b t) :
18 HasDerivAt (fun s => F (s, s)) (a + b) t := by
19 have h₁ := hF.hasFDerivAt.comp_hasDerivAt (f := fun s => (s, t)) t
20 ((hasDerivAt_id t).prodMk (hasDerivAt_const t t))
21 have h₂ := hF.hasFDerivAt.comp_hasDerivAt (f := fun s => (t, s)) t
22 ((hasDerivAt_const t t).prodMk (hasDerivAt_id t))
23 have hd := hF.hasFDerivAt.comp_hasDerivAt (f := fun s => (s, s)) t
24 ((hasDerivAt_id t).prodMk (hasDerivAt_id t))
25 have h₁' : fderiv ℝ F (t, t) (1, 0) = a := h₁.unique ha
26 have h₂' : fderiv ℝ F (t, t) (0, 1) = b := h₂.unique hb
27 exact hd.congr_deriv (by
28 rw [show ((1 : ℝ), (1 : ℝ)) = (1, 0) + (0, 1) by simp, map_add, h₁', h₂'])

35/-- **Math.** Joint differentiability of the actual moving-point metric pairing follows
36from joint smoothness of its chart Gram entries and differentiability of the
37curve and tangent fields in their actual tangent-bundle trivialization. -/
38theorem differentiableAt_metricInner_time_curve
39 {g : ℝ → Riemannian.RiemannianMetric I M} {J : Set ℝ}
40 (hg : IsSmoothMetricFamilyOn g J) {t : ℝ} (ht : t ∈ interior J)
41 {γ : ℝ → M} {V W : ∀ r, TangentSpace I (γ r)} {DV DW : E}
42 (hV : HasCovDerivAlongAt (I := I) (g t) γ V t DV)
43 (hW : HasCovDerivAlongAt (I := I) (g t) γ W t DW) :
44 DifferentiableAt ℝ
45 (fun z : ℝ × ℝ => (g z.1).metricInner (γ z.2) (V z.2) (W z.2)) (t, t) := by
46 classical
47 obtain ⟨hmem, v, dV, hu, hv, _⟩ := hV
48 obtain ⟨_, _, dW, _, hw, _⟩ := hW
49 let u := chartLocalCurve (I := I) γ t
50 let vcoord := chartFieldCoord (I := I) (γ t) γ V
51 let wcoord := chartFieldCoord (I := I) (γ t) γ W
52 have hsrc : γ t ∈ (extChartAt I (γ t)).source := by
53 rw [extChartAt_source]
54 exact mem_chart_source H (γ t)
55 have htarget : u t ∈ (extChartAt I (γ t)).target :=
56 (extChartAt I (γ t)).map_source hsrc
57 have hmap : DifferentiableAt ℝ (fun z : ℝ × ℝ => (z.1, u z.2)) (t, t) :=
58 differentiableAt_fst.prodMk (hu.differentiableAt.comp (f := fun z : ℝ × ℝ => z.2) (t, t) differentiableAt_snd)
59 have hgram (i j : Fin (Module.finrank ℝ E)) :
60 DifferentiableAt ℝ
61 (fun z : ℝ × ℝ => chartGramOnE (I := I) (g z.1) (γ t) i j (u z.2))
62 (t, t) :=
63 ((contDiffAt_chartGramOnE_timeSpace hg (γ t) i j ht htarget).differentiableAt
64 (by norm_num)).comp (f := fun z : ℝ × ℝ => (z.1, u z.2))
65 (g := fun z : ℝ × E => chartGramOnE (I := I) (g z.1) (γ t) i j z.2) (t, t) hmap
66 have hvc (i : Fin (Module.finrank ℝ E)) : DifferentiableAt ℝ
67 (fun z : ℝ × ℝ => chartCoord (E := E) i (vcoord z.2)) (t, t) := by
68 exact (chartCoordFunctional (E := E) i).differentiableAt.comp (t, t)
69 (hv.differentiableAt.comp (f := fun z : ℝ × ℝ => z.2) (t, t) differentiableAt_snd)
70 have hwc (j : Fin (Module.finrank ℝ E)) : DifferentiableAt ℝ
71 (fun z : ℝ × ℝ => chartCoord (E := E) j (wcoord z.2)) (t, t) := by
72 exact (chartCoordFunctional (E := E) j).differentiableAt.comp (t, t)
73 (hw.differentiableAt.comp (f := fun z : ℝ × ℝ => z.2) (t, t) differentiableAt_snd)
74 have hchart : DifferentiableAt ℝ
75 (fun z : ℝ × ℝ => chartMetricInner (I := I) (g z.1) (γ t)
76 (u z.2) (vcoord z.2) (wcoord z.2)) (t, t) := by
77 simp only [chartMetricInner_def]
78 exact DifferentiableAt.fun_sum fun i _ =>
79 DifferentiableAt.fun_sum fun j _ => ((hgram i j).mul (hvc i)).mul (hwc j)
80 apply hchart.congr_of_eventuallyEq
81 have hmem' : ∀ᶠ z : ℝ × ℝ in 𝓝 (t, t), γ z.2 ∈ (chartAt H (γ t)).source :=
82 (show Tendsto (fun z : ℝ × ℝ => z.2) (𝓝 (t, t)) (𝓝 t) from
83 continuousAt_snd.tendsto).eventually hmem
84 filter_upwards [hmem'] with z hz
85 exact metricInner_eq_chartMetricInner (I := I) (g z.1) (γ t) hz (V z.2) (W z.2)

87/-- **Math.** The moving-metric product rule. The covariant derivatives are taken for
88`g t` at the evaluation time. The additional term is the prescribed, genuine
89pointwise time variation of the metric, not a postulated moving-curve identity. -/
90theorem hasDerivAt_movingMetricInner
91 {g : ℝ → Riemannian.RiemannianMetric I M}
92 {h : ℝ → ∀ p : M, TangentSpace I p → TangentSpace I p → ℝ}
93 {J : Set ℝ} (hg : IsSmoothMetricFamilyOn g J)
94 (hh : IsMetricVariationOn g h J) {t : ℝ} (ht : t ∈ interior J)
95 {γ : ℝ → M} {V W : ∀ r, TangentSpace I (γ r)} {DV DW : E}
96 (hV : HasCovDerivAlongAt (I := I) (g t) γ V t DV)
97 (hW : HasCovDerivAlongAt (I := I) (g t) γ W t DW) :
98 HasDerivAt (fun s => (g s).metricInner (γ s) (V s) (W s))
99 (h t (γ t) (V t) (W t) +
100 ((g t).metricInner (γ t) DV (W t) + (g t).metricInner (γ t) (V t) DW)) t := by
101 exact hasDerivAt_diagonal_of_slices
102 (differentiableAt_metricInner_time_curve hg ht hV hW)
103 ((hh t (interior_subset ht) (γ t) (V t) (W t)).hasDerivAt
104 (mem_interior_iff_mem_nhds.mp ht))
105 (hV.hasDerivAt_metricInner hW)

107/-- **Math.** Squared length of an actual tangent field along a moving curve
108under Ricci flow. This is the metric-evolution contribution used in the speed
109and curvature calculations. -/
110theorem hasDerivAt_ricciFlow_field_sq [SigmaCompactSpace M] [T2Space M]
111 {g : ℝ → Riemannian.RiemannianMetric I M} {J : Set ℝ}
112 (hflow : IsRicciFlowOn g J) {t : ℝ} (ht : t ∈ interior J)
113 {γ : ℝ → M} {V : ∀ r, TangentSpace I (γ r)} {DV : E}
114 (hV : HasCovDerivAlongAt (I := I) (g t) γ V t DV) :
115 HasDerivAt (fun s => (g s).metricInner (γ s) (V s) (V s))
116 (-2 * ricciTensorAt (g t) (γ t) (V t) (V t) +
117 2 * (g t).metricInner (γ t) DV (V t)) t := by
118 have hd := hasDerivAt_movingMetricInner hflow.smooth
119 (isMetricVariationOn_of_isRicciFlowOn hflow) ht hV hV
120 apply hd.congr_deriv
121 rw [(g t).metricInner_comm (γ t) (V t) DV]
