15/-- Spatial differentiation preserves joint smoothness on an arbitrary time set. -/
16theorem contDiffOn_spatialFDeriv_apply
17 {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
18 {J : Set ℝ} {U : Set E} (hU : IsOpen U) {F : ℝ × E → ℝ}
19 (hF : ContDiffOn ℝ ∞ F (J ×ˢ U)) (v : E) :
20 ContDiffOn ℝ ∞
21 (fun z : ℝ × E => fderiv ℝ (fun y => F (z.1, y)) z.2 v)
22 (J ×ˢ U) := by
23 let S : Set (ℝ × E) := J ×ˢ U
24 let P : (ℝ × E) → E → ℝ := fun z y => F (z.1, y)
25 let gs : (ℝ × E) → E := fun z => z.2
26 let ks : (ℝ × E) → E := fun _ => v
27 have hP : ContDiffOn ℝ ∞ (Function.uncurry P) (S ×ˢ U) := by
28 change ContDiffOn ℝ ∞
29 (fun w : (ℝ × E) × E => F (w.1.1, w.2)) (S ×ˢ U)
30 have hmap : ContDiffOn ℝ ∞
31 (fun w : (ℝ × E) × E => (w.1.1, w.2)) (S ×ˢ U) :=
32 contDiffOn_fst.fst.prodMk contDiffOn_snd
33 exact hF.comp hmap (fun _ hw => ⟨hw.1.1, hw.2⟩)
34 intro z hz
35 have hgs : ContDiffWithinAt ℝ ∞ gs S z := contDiffWithinAt_snd
36 have hks : ContDiffWithinAt ℝ ∞ ks S z := contDiffWithinAt_const
37 have hpartial := (hP (z, gs z) ⟨hz, hz.2⟩).fderivWithin_apply
38 hgs hks hU.uniqueDiffOn
39 (show (∞ : ℕ∞ω) + 1 ≤ ∞ from le_rfl) hz (by
40 intro w hw
41 exact hw.2)
42 refine hpartial.congr_of_eventuallyEq_of_mem ?_ hz
43 filter_upwards [self_mem_nhdsWithin] with w hw
44 rw [fderivWithin_of_isOpen hU hw.2]

51/-- Covariant Ricci coefficients are jointly smooth on the whole time set. -/
52theorem contDiffOn_chartCovRicciOnE_timeSpace
53 {g : ℝ → Riemannian.RiemannianMetric I M} {J : Set ℝ}
54 (hg : IsSmoothMetricFamilyOn g J) (α : M)
55 (r i j : Fin (Module.finrank ℝ E)) :
56 ContDiffOn ℝ ∞
57 (fun z : ℝ × E => chartCovRicciOnE (I := I) (g z.1) α r i j z.2)
58 (J ×ˢ (extChartAt I α).target) := by
59 classical
60 have hRic := contDiffOn_chartRicciCoefOnE_timeSpace hg α
61 have hΓ := contDiffOn_chartChristoffel_timeSpace_on_timeSet hg α
62 have hpartial : ContDiffOn ℝ ∞
63 (fun z : ℝ × E => partialDeriv (E := E) r
64 (chartRicciCoefOnE (I := I) (g z.1) α i j) z.2)
65 (J ×ˢ (extChartAt I α).target) := by
66 simpa only [partialDeriv] using contDiffOn_spatialFDeriv_apply
67 (isOpen_extChartAt_target (I := I) α) (hRic i j) ((Module.finBasis ℝ E) r)
68 unfold chartCovRicciOnE
69 exact (hpartial.sub (ContDiffOn.sum fun s _ => (hΓ r i s).mul (hRic s j))).sub
70 (ContDiffOn.sum fun s _ => (hΓ r j s).mul (hRic i s))

85/-- Canonical intrinsic covariant derivative of Ricci, with its actual
86Levi-Civita connection and proved Levi-Civita property. -/
87def actualCovRicci (g : Riemannian.RiemannianMetric I M) (p : M)
88 (u v w : TangentSpace I p) : ℝ :=
89 covRicciAt g g.leviCivitaConnection
90 (g.leviCivitaConnection.isLeviCivita_of_koszulDual g
91 (fun X Y W q => g.koszulDualSection_dual X Y W q)) p u v w
