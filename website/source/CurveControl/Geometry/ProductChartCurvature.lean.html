4/-!
5# Actual chart curvature of the product metric
6
7The bilinear Christoffel map and its derivative split because the actual
8Christoffel contraction splits on an open chart target. Substitution in the
9upstream curvature definition proves the product curvature identity.
10The normed chart models need no inner-product-space structure.
11-/

24/-- **Math.** The upstream bilinear-map packaging agrees with its contraction,
25without requiring an inner-product norm on the chart model. -/
26theorem chartChristoffelBilin_apply_normed (g : RiemannianMetric I M) (a : M)
27 (y u v : E) : chartChristoffelBilin g a y u v = chartChristoffelContraction g a u v y := by
28 simp only [chartChristoffelBilin, chartChristoffelContraction,
29 sum_apply, ContinuousLinearMap.smulRight_apply, smul_apply, chartCoordFunctional_apply,
30 Finset.sum_smul, smul_smul]
31 conv_rhs => rw [Finset.sum_comm]
32 refine Finset.sum_congr rfl fun i _ => ?_
33 conv_rhs => rw [Finset.sum_comm]
34 refine Finset.sum_congr rfl fun j _ => ?_
35 refine Finset.sum_congr rfl fun k _ => ?_
36 congr 1
37 ring

39/-- **Math.** Smoothness of the actual bilinear Christoffel map on the interior
40chart target, valid for arbitrary finite-dimensional normed models. -/
41theorem contDiffOn_chartChristoffelBilin_normed (g : RiemannianMetric I M) (a : M) :
42 ContDiffOn ℝ ∞ (chartChristoffelBilin g a) (interior (extChartAt I a).target) := by
43 unfold chartChristoffelBilin
44 refine ContDiffOn.sum fun i _ => ContDiffOn.sum fun j _ => ContDiffOn.sum fun k _ => ?_
45 have hc : ContDiffOn ℝ ∞ (fun y => chartChristoffel g a i j k y • Module.finBasis ℝ E k)
46 (interior (extChartAt I a).target) :=
47 (chartChristoffel_contDiffOn_interior g a i j k).smul contDiffOn_const
48 exact contDiffOn_const.smulRight (contDiffOn_const.smulRight hc)

52/-- **Math.** Differentiability of the actual Christoffel bilinear map at any
53point of a boundaryless chart target. -/
54theorem differentiableAt_chartChristoffelBilin_normed (g : RiemannianMetric I M)
55 (a : M) (y : E) (hy : y ∈ (extChartAt I a).target) :
56 DifferentiableAt ℝ (chartChristoffelBilin g a) y := by
57 have hi : y ∈ interior (extChartAt I a).target :=
58 (isOpen_extChartAt_target (I := I) a).interior_eq.symm ▸ hy
59 exact ((contDiffOn_chartChristoffelBilin_normed g a).contDiffAt
60 (isOpen_interior.mem_nhds hi)).differentiableAt (by norm_num)

62/-- **Math.** Differentiate the actual Christoffel map along a coordinate line,
63then evaluate on two constant tangent-coordinate vectors. -/
64theorem chartChristoffelBilin_line_hasDerivAt (g : RiemannianMetric I M)
65 (a : M) (y x u v : E) (hy : y ∈ (extChartAt I a).target) :
66 HasDerivAt (fun r : ℝ => chartChristoffelBilin g a (y + r • x) u v)
67 (fderiv ℝ (chartChristoffelBilin g a) y x u v) 0 := by
68 have hline : HasDerivAt (fun r : ℝ => y + r • x) x 0 := by
69 simpa using ((hasDerivAt_id (0 : ℝ)).smul_const x).const_add y
70 have hG := (differentiableAt_chartChristoffelBilin_normed g a y hy).hasFDerivAt
71 have hGline : HasDerivAt (fun r : ℝ => chartChristoffelBilin g a (y + r • x))
72 (fderiv ℝ (chartChristoffelBilin g a) y x) 0 := by
73 have hg : HasFDerivAt (chartChristoffelBilin g a) (fderiv ℝ (chartChristoffelBilin g a) y)
74 ((fun r : ℝ => y + r • x) 0) := by simpa only [zero_smul, add_zero] using hG
75 exact HasFDerivAt.comp_hasDerivAt (l := chartChristoffelBilin g a)
76 (l' := fderiv ℝ (chartChristoffelBilin g a) y)
77 (f := fun r : ℝ => y + r • x) (f' := x) (0 : ℝ) hg hline
78 have hh := (hGline.clm_apply (hasDerivAt_const 0 u)).clm_apply (hasDerivAt_const 0 v)
79 simpa only [zero_smul, add_zero, map_zero] using hh

86/-- **Math.** Splitting of the actual bilinear Christoffel map. -/
87theorem chartChristoffelBilin_product (g : RiemannianMetric I M) (h : RiemannianMetric J N)
88 (a : M × N) (y u v : E × F) (hy : y ∈ (extChartAt (I.prod J) a).target) :
89 chartChristoffelBilin (productMetric g h) a y u v =
90 (chartChristoffelBilin g a.1 y.1 u.1 v.1, chartChristoffelBilin h a.2 y.2 u.2 v.2) := by
91 simp only [chartChristoffelBilin_apply_normed]
92 exact chartChristoffelContraction_product g h a y u v hy

94/-- **Math.** Derivative splitting follows by differentiating the actual
95Christoffel identity on the open product-chart target. -/
96theorem fderiv_chartChristoffelBilin_product
97 (g : RiemannianMetric I M) (h : RiemannianMetric J N)
98 (a : M × N) (y x u v : E × F) (hy : y ∈ (extChartAt (I.prod J) a).target) :
99 fderiv ℝ (chartChristoffelBilin (productMetric g h) a) y x u v =
100 (fderiv ℝ (chartChristoffelBilin g a.1) y.1 x.1 u.1 v.1,
101 fderiv ℝ (chartChristoffelBilin h a.2) y.2 x.2 u.2 v.2) := by
102 have hy' : y.1 ∈ (extChartAt I a.1).target ∧ y.2 ∈ (extChartAt J a.2).target := by
103 simpa only [extChartAt_prod, PartialEquiv.prod_target, Set.mem_prod] using hy
104 have hd := chartChristoffelBilin_line_hasDerivAt (productMetric g h) a y x u v hy
105 have hd1 := chartChristoffelBilin_line_hasDerivAt g a.1 y.1 x.1 u.1 v.1 hy'.1
106 have hd2 := chartChristoffelBilin_line_hasDerivAt h a.2 y.2 x.2 u.2 v.2 hy'.2
107 have hline : ContinuousAt (fun r : ℝ => y + r • x) 0 := by fun_prop
108 have hnear : ∀ᶠ r : ℝ in nhds 0, y + r • x ∈ (extChartAt (I.prod J) a).target :=
109 hline.eventually ((isOpen_extChartAt_target a).mem_nhds (by simpa using hy))
110 have heq : (fun r : ℝ => chartChristoffelBilin (productMetric g h) a (y + r • x) u v) =ᶠ[nhds 0]
111 (fun r : ℝ => (chartChristoffelBilin g a.1 (y.1 + r • x.1) u.1 v.1,
112 chartChristoffelBilin h a.2 (y.2 + r • x.2) u.2 v.2)) := by
113 filter_upwards [hnear] with r hr
114 exact chartChristoffelBilin_product g h a (y + r • x) u v hr
115 exact hd.unique ((hd1.prodMk hd2).congr_of_eventuallyEq heq)

117/-- **Math.** Actual product chart Riemann curvature splits into the two factor
118curvatures. All Christoffel and derivative splitting terms have been proved. -/
119theorem chartCurvature_product (g : RiemannianMetric I M) (h : RiemannianMetric J N)
120 (a : M × N) (y x u v : E × F) (hy : y ∈ (extChartAt (I.prod J) a).target) :
121 chartCurvature (productMetric g h) a y x u v =
122 (chartCurvature g a.1 y.1 x.1 u.1 v.1,
123 chartCurvature h a.2 y.2 x.2 u.2 v.2) := by
124 unfold chartCurvature christoffelCurvature
125 simp only [fderiv_chartChristoffelBilin_product g h a y _ _ _ hy,
126 chartChristoffelBilin_product g h a y _ _ hy]
127 rfl
