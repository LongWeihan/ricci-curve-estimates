4/-!
5# Product chart connection splitting
6
7The actual tangent coordinate change and inverse trivialization split on a
8product chart. Hence the genuine chart Gram form of `productMetric` splits.
9Differentiating that identity along coordinate lines proves metric compatibility
10of the factorwise Christoffel candidate. Its symmetry and a pointwise Koszul
11uniqueness argument identify it with the product metric's actual Christoffel
12contraction throughout the chart target. No connection-splitting assumption is
13used. Models need only be finite-dimensional normed spaces: the max-norm product
14does not carry a spurious inner-product-space instance. Identification with a
15canonical intrinsic LC connection on an L²-remodelled manifold is separate.
16-/

25/-- **Math.** Pointwise uniqueness of a symmetric connection coefficient with
26a fixed metric-compatibility identity. This is the algebraic Koszul argument,
27independent of any preferred basis or Hilbert norm on the chart model. -/
28theorem connectionCoefficient_unique {V : Type*}
29 (G : V → V → ℝ) (A B : V → V → V)
30 (hGsym : ∀ u v, G u v = G v u)
31 (hGsep : ∀ u v, (∀ w, G u w = G v w) → u = v)
32 (hAsym : ∀ u v, A u v = A v u) (hBsym : ∀ u v, B u v = B v u)
33 (hbalance : ∀ u v w, G (A u v) w + G v (A u w) =
34 G (B u v) w + G v (B u w)) : A = B := by
35 funext u v
36 apply hGsep
37 intro w
38 have h1 := hbalance u v w
39 have h2 := hbalance v u w
40 have h3 := hbalance w u v
41 rw [hGsym v (A u w), hGsym v (B u w)] at h1
42 rw [hAsym v u, hBsym v u, hGsym u (A v w), hGsym u (B v w)] at h2
43 rw [hAsym w u, hBsym w u, hAsym w v, hBsym w v,
44 hGsym u (A v w), hGsym u (B v w)] at h3
45 linarith

55/-- **Math.** Actual tangent-coordinate changes of the product manifold split. -/
56theorem tangentCoordChange_product (a b p : M × N)
57 (ha : p ∈ (extChartAt (I.prod J) a).source)
58 (hb : p ∈ (extChartAt (I.prod J) b).source) :
59 tangentCoordChange (I.prod J) a b p =
60 (tangentCoordChange I a.1 b.1 p.1).prodMap (tangentCoordChange J a.2 b.2 p.2) := by
61 have ha' : p.1 ∈ (extChartAt I a.1).source ∧ p.2 ∈ (extChartAt J a.2).source := by
62 simpa only [extChartAt_prod, PartialEquiv.prod_source, Set.mem_prod] using ha
63 have hb' : p.1 ∈ (extChartAt I b.1).source ∧ p.2 ∈ (extChartAt J b.2).source := by
64 simpa only [extChartAt_prod, PartialEquiv.prod_source, Set.mem_prod] using hb
65 have h1 : HasFDerivAt (extChartAt I b.1 ∘ (extChartAt I a.1).symm)
66 (tangentCoordChange I a.1 b.1 p.1) (extChartAt I a.1 p.1) := by
67 simpa only [I.range_eq_univ, hasFDerivWithinAt_univ] using
68 hasFDerivWithinAt_tangentCoordChange (I := I) ⟨ha'.1, hb'.1⟩
69 have h2 : HasFDerivAt (extChartAt J b.2 ∘ (extChartAt J a.2).symm)
70 (tangentCoordChange J a.2 b.2 p.2) (extChartAt J a.2 p.2) := by
71 simpa only [J.range_eq_univ, hasFDerivWithinAt_univ] using
72 hasFDerivWithinAt_tangentCoordChange (I := J) ⟨ha'.2, hb'.2⟩
73 rw [tangentCoordChange_def, ModelWithCorners.range_eq_univ, fderivWithin_univ]
74 convert (HasFDerivAt.prodMap (extChartAt I a.1 p.1, extChartAt J a.2 p.2) h1 h2).fderiv using 1
75 simp only [extChartAt_prod, PartialEquiv.prod_coe_symm, PartialEquiv.prod_coe,
76 Function.comp_def]
77 rfl

79/-- **Math.** The inverse tangent-bundle trivialization of the actual product
80chart acts separately on the two tangent components. -/
81theorem trivializationAt_symm_product (a p : M × N)
82 (hp : p ∈ (chartAt (ModelProd H K) a).source) (u : E × F) :
83 (trivializationAt (E × F) (TangentSpace (I.prod J)) a).symm p u =
84 ((trivializationAt E (TangentSpace I) a.1).symm p.1 u.1,
85 (trivializationAt F (TangentSpace J) a.2).symm p.2 u.2) := by
86 have hp' : p.1 ∈ (chartAt H a.1).source ∧ p.2 ∈ (chartAt K a.2).source := hp
87 rw [trivializationAt_symm_eq_tangentCoordChange a hp,
88 tangentCoordChange_product a p p (by rwa [extChartAt_source]) (mem_extChartAt_source p),
89 trivializationAt_symm_eq_tangentCoordChange a.1 hp'.1,
90 trivializationAt_symm_eq_tangentCoordChange a.2 hp'.2]
91 rfl

95/-- **Math.** Metric compatibility along a straight coordinate line, without an
96inner-product-space instance on the chart model. -/
97theorem chartMetricInner_line_hasDerivAt (g : RiemannianMetric I M) (a : M)
98 (y x u v : E) (hy : y ∈ (extChartAt I a).target) :
99 HasDerivAt (fun r : ℝ => chartMetricInner g a (y + r • x) u v)
100 (chartMetricInner g a y (chartChristoffelContraction g a x u y) v +
101 chartMetricInner g a y u (chartChristoffelContraction g a x v y)) 0 := by
102 have hline : HasDerivAt (fun r : ℝ => y + r • x) x 0 := by
103 simpa using ((hasDerivAt_id (0 : ℝ)).smul_const x).const_add y
104 have hbase : (extChartAt I a).symm y ∈
105 (trivializationAt E (TangentSpace I) a).baseSet := by
106 simpa only [extChartAt_source, TangentBundle.trivializationAt_baseSet] using (extChartAt I a).map_target hy
107 have hG : ∀ i j, DifferentiableAt ℝ (chartGramOnE g a i j) y := by
108 intro i j
109 exact ((chartGramOnE_contDiffOn g a i j).contDiffAt
110 ((isOpen_extChartAt_target a).mem_nhds hy)).differentiableAt (by norm_num)
111 have hd := hasDerivAt_chartMetricInner_along g a (fun r => y + r • x)
112 (fun _ => u) (fun _ => v) hline.differentiableAt (differentiableAt_const u)
113 (differentiableAt_const v) (by simpa using hG) (by simpa using hbase)
114 simpa only [zero_smul, add_zero, covariantDerivCoord, hline.deriv,
115 deriv_const, zero_add] using hd

117/-- **Math.** The chart Gram form of the product metric splits, with vectors
118read through the genuine chart trivializations. -/
119theorem chartMetricInner_product (g : RiemannianMetric I M) (h : RiemannianMetric J N)
120 (a p : M × N) (hp : p ∈ (chartAt (ModelProd H K) a).source) (u v : E × F) :
121 chartMetricInner (productMetric g h) a (extChartAt (I.prod J) a p) u v =
122 chartMetricInner g a.1 (extChartAt I a.1 p.1) u.1 v.1 +
123 chartMetricInner h a.2 (extChartAt J a.2 p.2) u.2 v.2 := by
124 have hp' : p.1 ∈ (chartAt H a.1).source ∧ p.2 ∈ (chartAt K a.2).source := hp

127 productMetric_inner,
128 chartMetricInner_extChartAt_eq_metricInner _ _ hp'.1,

131/-- **Math.** The product chart metric splitting at every point of its target. -/
132theorem chartMetricInner_product_target (g : RiemannianMetric I M) (h : RiemannianMetric J N)
133 (a : M × N) (y u v : E × F) (hy : y ∈ (extChartAt (I.prod J) a).target) :
134 chartMetricInner (productMetric g h) a y u v =
135 chartMetricInner g a.1 y.1 u.1 v.1 + chartMetricInner h a.2 y.2 u.2 v.2 := by
136 have hp : (extChartAt (I.prod J) a).symm y ∈ (chartAt (ModelProd H K) a).source := by
137 simpa only [extChartAt_source] using (extChartAt (I.prod J) a).map_target hy
138 have he := chartMetricInner_product g h a ((extChartAt (I.prod J) a).symm y) hp u v
139 rw [(extChartAt (I.prod J) a).right_inv hy] at he
140 have hy' : y.1 ∈ (extChartAt I a.1).target ∧ y.2 ∈ (extChartAt J a.2).target := by
141 simpa only [extChartAt_prod, PartialEquiv.prod_target, Set.mem_prod] using hy
142 simpa only [extChartAt_prod, PartialEquiv.prod_coe_symm, Prod.map_fst, Prod.map_snd,
143 (extChartAt I a.1).right_inv hy'.1, (extChartAt J a.2).right_inv hy'.2] using he

146/-- **Math.** Nondegeneracy of the actual chart Gram form, transported through
147the invertible tangent-coordinate change. -/
148theorem chartMetricInner_separates (g : RiemannianMetric I M) (a : M)
149 (y : E) (hy : y ∈ (extChartAt I a).target) (u v : E)
150 (huv : ∀ w, chartMetricInner g a y u w = chartMetricInner g a y v w) : u = v := by
151 let p := (extChartAt I a).symm y
152 have hp : p ∈ (extChartAt I a).source := (extChartAt I a).map_target hy
153 have hpc : p ∈ (chartAt H a).source := by simpa only [extChartAt_source] using hp
154 have hpp := mem_extChartAt_source (I := I) p
155 have hread : ∀ z w, chartMetricInner g a y z w =
156 g.metricInner p (tangentCoordChange I a p p z) (tangentCoordChange I a p p w) := by
157 intro z w
158 rw [← (extChartAt I a).right_inv hy]
159 rw [chartMetricInner_extChartAt_eq_metricInner _ _ hpc,
160 trivializationAt_symm_eq_tangentCoordChange a hpc,
161 trivializationAt_symm_eq_tangentCoordChange a hpc]
162 have hTv : tangentCoordChange I a p p u = tangentCoordChange I a p p v := by
163 apply (g.metricInner_eq_iff_eq p _ _).mp
164 intro z
165 have hh := huv (tangentCoordChange I p a p z)
166 simp only [hread] at hh
167 rw [tangentCoordChange_comp (I := I) ⟨⟨hpp, hp⟩, hpp⟩,
168 tangentCoordChange_self (I := I) hpp] at hh
169 exact hh
170 have hh := congrArg (tangentCoordChange I p a p) hTv
171 rw [tangentCoordChange_comp (I := I) ⟨⟨hp, hpp⟩, hp⟩,
172 tangentCoordChange_comp (I := I) ⟨⟨hp, hpp⟩, hp⟩,
173 tangentCoordChange_self (I := I) hp, tangentCoordChange_self (I := I) hp] at hh
174 exact hh

176/-- **Math.** Differentiating the genuine product chart metric splitting shows
177that the two factor Christoffel corrections obey the product metric identity. -/
178theorem product_connection_metric_balance
179 (g : RiemannianMetric I M) (h : RiemannianMetric J N)
180 (a : M × N) (y x u v : E × F) (hy : y ∈ (extChartAt (I.prod J) a).target) :
181 chartMetricInner (productMetric g h) a y
182 (chartChristoffelContraction (productMetric g h) a x u y) v +
183 chartMetricInner (productMetric g h) a y u
184 (chartChristoffelContraction (productMetric g h) a x v y) =
185 chartMetricInner (productMetric g h) a y
186 (chartChristoffelContraction g a.1 x.1 u.1 y.1,
187 chartChristoffelContraction h a.2 x.2 u.2 y.2) v +
188 chartMetricInner (productMetric g h) a y u
189 (chartChristoffelContraction g a.1 x.1 v.1 y.1,
190 chartChristoffelContraction h a.2 x.2 v.2 y.2) := by
191 have hy' : y.1 ∈ (extChartAt I a.1).target ∧ y.2 ∈ (extChartAt J a.2).target := by
192 simpa only [extChartAt_prod, PartialEquiv.prod_target, Set.mem_prod] using hy
193 have hd := chartMetricInner_line_hasDerivAt (productMetric g h) a y x u v hy
194 have hd1 := chartMetricInner_line_hasDerivAt g a.1 y.1 x.1 u.1 v.1 hy'.1
195 have hd2 := chartMetricInner_line_hasDerivAt h a.2 y.2 x.2 u.2 v.2 hy'.2
196 have hline : ContinuousAt (fun r : ℝ => y + r • x) 0 := by fun_prop
197 have hnear : ∀ᶠ r : ℝ in nhds 0, y + r • x ∈ (extChartAt (I.prod J) a).target :=
198 hline.eventually ((isOpen_extChartAt_target a).mem_nhds (by simpa using hy))
199 have heq : (fun r : ℝ => chartMetricInner (productMetric g h) a (y + r • x) u v) =ᶠ[nhds 0]
200 (fun r : ℝ => chartMetricInner g a.1 (y.1 + r • x.1) u.1 v.1 +
201 chartMetricInner h a.2 (y.2 + r • x.2) u.2 v.2) := by
202 filter_upwards [hnear] with r hr
203 exact chartMetricInner_product_target g h a (y + r • x) u v hr
204 have he := hd.unique ((hd1.add hd2).congr_of_eventuallyEq heq)
205 conv_rhs => rw [chartMetricInner_product_target g h a y _ _ hy,
206 chartMetricInner_product_target g h a y _ _ hy]
207 dsimp only at he ⊢
208 linarith

210/-- **Math.** Actual chart Christoffel coefficients of the product metric split
211on the product chart target. This is derived from the differentiated metric and
212torsion-free symmetry; no connection-splitting hypothesis is used. -/
213theorem chartChristoffelContraction_product
214 (g : RiemannianMetric I M) (h : RiemannianMetric J N)
215 (a : M × N) (y x u : E × F) (hy : y ∈ (extChartAt (I.prod J) a).target) :
216 chartChristoffelContraction (productMetric g h) a x u y =
217 (chartChristoffelContraction g a.1 x.1 u.1 y.1,
218 chartChristoffelContraction h a.2 x.2 u.2 y.2) := by
219 have hsym : ∀ v w, chartMetricInner (productMetric g h) a y v w =
220 chartMetricInner (productMetric g h) a y w v := by
221 intro v w
222 unfold chartMetricInner
223 rw [Finset.sum_comm]
224 apply Finset.sum_congr rfl
225 intro i _
226 apply Finset.sum_congr rfl
227 intro j _
228 rw [chartGramOnE_symm (productMetric g h) a j i]
229 ring
230 have hh := connectionCoefficient_unique
231 (chartMetricInner (productMetric g h) a y)
232 (fun v w => chartChristoffelContraction (productMetric g h) a v w y)
233 (fun v w => (chartChristoffelContraction g a.1 v.1 w.1 y.1,
234 chartChristoffelContraction h a.2 v.2 w.2 y.2)) hsym
235 (chartMetricInner_separates (productMetric g h) a y hy)
236 (fun v w => chartChristoffelContraction_symm (productMetric g h) a v w y)
237 (fun v w => Prod.ext (chartChristoffelContraction_symm g a.1 v.1 w.1 y.1)
238 (chartChristoffelContraction_symm h a.2 v.2 w.2 y.2))
239 (fun v w z => product_connection_metric_balance g h a y v w z hy)
240 exact congrFun (congrFun hh x) u
