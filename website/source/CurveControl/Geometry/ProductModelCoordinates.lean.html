102theorem productL2_chartMetricInner_on_target

104 (a : M × N) (y u v : WithLp 2 (E × F))
105 (hy : y ∈ (extChartAt (productL2Model I J) a).target) :
106 chartMetricInner (productL2Metric I J g h) a y u v =

110 have hp : p ∈ (chartAt (ModelProd H K) a).source := by
